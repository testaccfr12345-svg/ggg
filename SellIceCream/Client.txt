local sounds = {}
for key, id in pairs(Config.Sounds) do
	if id ~= "" then
		local sound = Instance.new("Sound")
		sound.Name = "IceCream_" .. key
		sound.SoundId = id
		sound.Volume = 0.6
		sound.Looped = key == "DayMusic" or key == "NightAmbience"
		sound.Parent = SoundService
		sounds[key] = sound
	end
end

local function play(key, speed)
	local sound = sounds[key]
	if not sound then return end
	sound.PlaybackSpeed = speed or 1
	if not sound.Looped then sound.TimePosition = 0 end
	sound:Play()
end

local function stopSound(key)
	if sounds[key] then sounds[key]:Stop() end
end

local function new(className, props, parent)
	local instance = Instance.new(className)
	for key, value in pairs(props) do instance[key] = value end
	instance.Parent = parent
	return instance
end

local function tween(object, time, props, style, direction)
	local t = TweenService:Create(object, TweenInfo.new(time, style or Enum.EasingStyle.Quad, direction or Enum.EasingDirection.Out), props)
	t:Play()
	return t
end

local function corner(parent, radius)
	return new("UICorner", { CornerRadius = radius == "round" and UDim.new(1, 0) or UDim.new(0, radius) }, parent)
end

local function money(n)
	n = math.floor(n or 0)
	local negative = n < 0
	n = math.abs(n)
	local text
	if n >= 1000000 then
		text = string.format("%.1fM", n / 1000000)
	elseif n >= 10000 then
		text = string.format("%.1fK", n / 1000)
	else
		text = tostring(n)
	end
	return (negative and "-$" or "$") .. text
end

local function text(parent, str, size, position, textSize, color, thickness, align)
	local label = new("TextLabel", {
		BackgroundTransparency = 1,
		Size = size,
		Position = position,
		Text = str,
		TextSize = textSize,
		TextColor3 = color or WHITE,
		Font = FONT,
		TextXAlignment = align or Enum.TextXAlignment.Center,
		ZIndex = 6,
	}, parent)
	if thickness ~= 0 then
		new("UIStroke", { Color = OUTLINE, Thickness = thickness or 2.5, LineJoinMode = Enum.LineJoinMode.Round }, label)
	end
	return label
end

local function chunky(parent, size, position, anchor, color, radius, depth)
	radius = radius or 14
	depth = depth or 6
	local holder = new("Frame", { Size = size, Position = position, AnchorPoint = anchor or Vector2.new(0, 0), BackgroundTransparency = 1 }, parent)
	local edge = new("Frame", { Size = UDim2.fromScale(1, 1), BackgroundColor3 = color:Lerp(BLACK, 0.35), BorderSizePixel = 0, ZIndex = 2 }, holder)
	corner(edge, radius)
	new("UIStroke", { Color = OUTLINE, Thickness = 3, ApplyStrokeMode = Enum.ApplyStrokeMode.Border }, edge)
	local face = new("Frame", { Size = UDim2.new(1, 0, 1, -depth), BackgroundColor3 = color, BorderSizePixel = 0, ZIndex = 3 }, holder)
	corner(face, radius)
	new("UIGradient", { Rotation = 90, Color = ColorSequence.new(WHITE, Color3.fromRGB(215, 215, 215)) }, face)
	local gloss = new("Frame", { Size = UDim2.new(1, -14, 0.34, 0), Position = UDim2.fromOffset(7, 4), BackgroundColor3 = WHITE, BackgroundTransparency = 0.78, BorderSizePixel = 0, ZIndex = 4 }, face)
	corner(gloss, radius == "round" and "round" or math.max(4, radius - 5))
	local piece = { holder = holder, edge = edge, face = face, depth = depth, color = color }
	piece.setColor = function(newColor)
		piece.color = newColor
		face.BackgroundColor3 = newColor
		edge.BackgroundColor3 = newColor:Lerp(BLACK, 0.35)
	end
	return piece
end

local function clickable(piece, callback)
	local button = new("TextButton", { Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, Text = "", AutoButtonColor = false, ZIndex = 20 }, piece.holder)
	local scale = new("UIScale", { Scale = 1 }, piece.holder)
	piece.scale = scale
	local function release()
		tween(piece.face, 0.12, { Position = UDim2.fromOffset(0, 0) }, Enum.EasingStyle.Back)
	end
	button.MouseButton1Down:Connect(function()
		tween(piece.face, 0.05, { Position = UDim2.fromOffset(0, piece.depth - 2) })
	end)
	button.MouseButton1Up:Connect(release)
	button.MouseLeave:Connect(function()
		release()
		tween(scale, 0.15, { Scale = 1 })
	end)
	button.MouseEnter:Connect(function()
		if UserInputService.MouseEnabled then tween(scale, 0.15, { Scale = 1.05 }, Enum.EasingStyle.Back) end
	end)
	button.Activated:Connect(function()
		play("Click")
		callback()
	end)
	piece.button = button
	return piece
end

local function bump(object, amount)
	local scale = object:FindFirstChildOfClass("UIScale") or new("UIScale", { Scale = 1 }, object)
	scale.Scale = amount or 1.15
	tween(scale, 0.3, { Scale = 1 }, Enum.EasingStyle.Back)
end

local function toast(message, kind)
	toastCount = toastCount + 1
	local frame = new("Frame", { Size = UDim2.fromOffset(440, 40), BackgroundColor3 = TOAST_COLORS[kind] or PINK, BorderSizePixel = 0, LayoutOrder = toastCount, ZIndex = 30 }, toasts)
	corner(frame, 14)

	local label = text(frame, message, UDim2.new(1, -16, 1, 0), UDim2.fromOffset(8, 0), 19, WHITE, 2.5)
	label.ZIndex = 31
	label.TextScaled = true
	new("UITextSizeConstraint", { MaxTextSize = 19, MinTextSize = 10 }, label)
	local scale = new("UIScale", { Scale = 0.4 }, frame)
	tween(scale, 0.3, { Scale = 1 }, Enum.EasingStyle.Back)
	task.delay(2.8, function()
		if not frame.Parent then return end
		tween(scale, 0.2, { Scale = 0 }, Enum.EasingStyle.Back, Enum.EasingDirection.In)
		task.wait(0.22)
		frame:Destroy()
	end)
	local list = {}
	for _, child in ipairs(toasts:GetChildren()) do
		if child:IsA("Frame") then table.insert(list, child) end
	end
	table.sort(list, function(a, b) return a.LayoutOrder < b.LayoutOrder end)
	if #list > 3 then list[1]:Destroy() end
end

local flavorButtons = {}

local function buildFlavorButtons()
	for _, piece in pairs(flavorButtons) do piece.holder:Destroy() end
	flavorButtons = {}
	local list = string.split(gameState:GetAttribute("ShopFlavors") or "", ",")
	for index, id in ipairs(list) do
		local flavor = FLAVOR_BY_ID[id]
		if flavor then
			local piece = chunky(flavorGrid, UDim2.fromOffset(96, 74), UDim2.fromOffset(0, 0), nil, flavor.color, 16, 6)
			piece.holder.LayoutOrder = index
			local dot = new("Frame", { Size = UDim2.fromOffset(26, 26), Position = UDim2.new(0.5, -13, 0, 6), BackgroundColor3 = flavor.color, BorderSizePixel = 0, ZIndex = 6 }, piece.face)
			corner(dot, "round")
			new("UIStroke", { Color = OUTLINE, Thickness = 2.5 }, dot)
			local name = text(piece.face, flavor.name, UDim2.new(1, -6, 0, 20), UDim2.fromOffset(3, 34), 17, WHITE, 2.5)
			name.TextScaled = true
			new("UITextSizeConstraint", { MaxTextSize = 17 }, name)
			clickable(piece, function()
				if serving then return end
				local count = 0
				for _, other in ipairs(currentScoops) do if other == id then count = count + 1 end end
				if #currentScoops >= Config.MaxScoopsTotal or count >= Config.MaxScoopsPerFlavor then
					toast("That's enough of that!", "bad")
					return
				end
				table.insert(currentScoops, id)
				refreshCone()
				bump(scoopHolder, 1.1)
			end)
			flavorButtons[id] = piece
		end
	end
end

local tutorialDone = {}
local hideTutorial

local function doServe()
	if serving then return end
	if #currentScoops == 0 then
		toast("Add scoops first!", "bad")
		return
	end
	if (gameState:GetAttribute("CurrentOrder") or "") == "" then
		toast("No customer at the counter yet!", "bad")
		return
	end
	serving = true
	local map = {}
	for _, id in ipairs(currentScoops) do map[id] = (map[id] or 0) + 1 end
	local upgrades = gameState:GetAttribute("ShopUpgrades") or ""
	local serveTime = string.find(upgrades, "FastScoop") and Config.FastServeTime or Config.ServeTime
	serveLabel.Text = "SCOOPING..."
	servePiece.setColor(GREY)
	task.wait(serveTime)
	ServeRemote:FireServer(map)
	currentScoops = {}
	refreshCone()
	serveLabel.Text = "SERVE"
	servePiece.setColor(GREEN)
	serving = false
	if hideTutorial then hideTutorial("flavors") end
end

local overlay = new("TextButton", { Size = UDim2.fromScale(1, 1), BackgroundColor3 = BLACK, BackgroundTransparency = 1, Text = "", AutoButtonColor = false, Visible = false, ZIndex = 50 }, gui)

local function closePanel()
	if not openPanel then return end
	local panel = openPanel
	openPanel = nil
	tween(overlay, 0.2, { BackgroundTransparency = 1 })
	tween(panel.scale, 0.18, { Scale = 0 }, Enum.EasingStyle.Back, Enum.EasingDirection.In).Completed:Connect(function()
		if openPanel ~= panel then panel.holder.Visible = false end
		if not openPanel then overlay.Visible = false end
	end)
end

showPanel = function(panel)
	if openPanel == panel then closePanel() return end
	if openPanel then openPanel.holder.Visible = false end
	openPanel = panel
	overlay.Visible = true
	panel.holder.Visible = true
	panel.scale.Scale = 0.4
	if panel.onOpen then panel.onOpen() end
	tween(panel.scale, 0.3, { Scale = uiScale.Scale }, Enum.EasingStyle.Back)
	tween(overlay, 0.2, { BackgroundTransparency = 0.5 })
end
overlay.Activated:Connect(closePanel)

local motorCache = setmetatable({}, { __mode = "k" })
local function motorsOf(model)
	local cached = motorCache[model]
	if cached then return cached end
	local found = {}
	for _, name in ipairs({ "LeftHip", "RightHip", "LeftShoulder", "RightShoulder", "Neck", "Waist", "LeftKnee", "RightKnee" }) do
		found[name] = model:FindFirstChild(name, true)
	end
	local head = model:FindFirstChild("Head")
	local faceGui = head and head:FindFirstChild("FaceGui")
	found.eyes = faceGui and { faceGui:FindFirstChild("EyeL"), faceGui:FindFirstChild("EyeR") } or {}
	found.blinkAt = os.clock() + math.random() * 4
	found.offset = math.random() * 10
	motorCache[model] = found
	return found
end

local function animateModel(model, t, monster)
	local rootPart = model:FindFirstChild("HumanoidRootPart")
	if not rootPart then return end
	local m = motorsOf(model)
	local speed = Vector3.new(rootPart.AssemblyLinearVelocity.X, 0, rootPart.AssemblyLinearVelocity.Z).Magnitude
	local walk = math.clamp(speed / 12, 0, 1)
	local still = model:GetAttribute("Still") == true
	local phase = t * (monster and 13 or 10) + m.offset
	local swing = math.sin(phase) * 0.7 * walk
	local idle = still and 0 or math.sin(t * 2 + m.offset) * 0.04
	if m.LeftHip then m.LeftHip.Transform = CFrame.Angles(swing, 0, 0) end
	if m.RightHip then m.RightHip.Transform = CFrame.Angles(-swing, 0, 0) end
	if m.LeftShoulder then m.LeftShoulder.Transform = CFrame.Angles(-swing * 0.8 + (monster and 0.3 or 0), 0, idle) end
	if m.RightShoulder then m.RightShoulder.Transform = CFrame.Angles(swing * 0.8 + (monster and 0.3 or 0), 0, -idle) end
	if m.Waist then m.Waist.Transform = CFrame.Angles(idle + (monster and 0.15 or 0), 0, 0) end
	if m.Neck then
		local neck = CFrame.new()
		if model:GetAttribute("HeadFollow") then
			local character = player.Character
			local myHead = character and character:FindFirstChild("Head")
			if myHead then
				local rel = rootPart.CFrame:PointToObjectSpace(myHead.Position)
				local yaw = math.clamp(math.atan2(-rel.X, -rel.Z), -1.3, 1.3)
				neck = CFrame.Angles(0, yaw, 0)
			end
		elseif monster then
			if math.random() < 0.04 then
				neck = CFrame.Angles(math.random() * 0.8 - 0.4, math.random() * 1.2 - 0.6, math.random() * 0.8 - 0.4)
				m.twitch = neck
				m.twitchUntil = t + 0.15
			elseif m.twitchUntil and t < m.twitchUntil then
				neck = m.twitch
			else
				neck = CFrame.Angles(0.25, 0, 0)
			end
		elseif not still then
			neck = CFrame.Angles(math.sin(t * 1.3 + m.offset) * 0.05, math.sin(t * 0.7 + m.offset) * 0.15, 0)
		end
		m.Neck.Transform = neck
	end
	if not still and not monster and #m.eyes == 2 and m.eyes[1] and m.eyes[2] then
		local blinking = t >= m.blinkAt and t < m.blinkAt + 0.12
		if t >= m.blinkAt + 0.12 then m.blinkAt = t + 2.5 + math.random() * 3 end
		for _, eye in ipairs(m.eyes) do
			eye.Size = blinking and UDim2.fromScale(0.24, 0.03) or UDim2.fromScale(0.24, 0.24)
			eye.Position = UDim2.new(eye.Position.X.Scale, 0, blinking and 0.41 or 0.3, 0)
		end
	end
end

	local t = os.clock()
	local phase = gameState:GetAttribute("Phase") or "Waiting"
	local endsAt = gameState:GetAttribute("EndsAt") or 0
	local left = math.max(0, math.floor(endsAt - workspace:GetServerTimeNow()))
	local clock = string.format("%d:%02d", math.floor(left / 60), left % 60)
	local day = gameState:GetAttribute("Day") or 1
	if phase == "Day" then
		phaseText.Text = "DAY " .. day .. " — " .. clock
	elseif phase == "Night" then
		phaseText.Text = "NIGHT " .. day .. " — " .. clock
	elseif phase == "Morning" then
		phaseText.Text = "GOOD MORNING!"
	else
		phaseText.Text = "OPENING..."
	end
	if phase == "Night" then
		for _, name in ipairs({ "Front", "Left", "Right", "Back" }) do
			local row = entryRows[name]
			local entryState = gameState:GetAttribute("Entry_" .. name .. "_State") or ""
			local health = gameState:GetAttribute("Entry_" .. name .. "_Health") or 0
			local maxHealth = math.max(1, gameState:GetAttribute("Entry_" .. name .. "_Max") or 100)
			local protected = entryState == "Locked" or entryState == "Boarded"
			local ratio = protected and math.clamp(health / maxHealth, 0, 1) or 0
			row.fill.Size = UDim2.fromScale(ratio, 1)
			row.fill.BackgroundColor3 = ratio > 0.5 and GREEN or (ratio > 0.25 and YELLOW or RED)
			local labels = { Open = "OPEN!", Closed = "UNLOCKED", Locked = "LOCKED", Glass = "NO BOARDS", Boarded = "BOARDED", Broken = "BROKEN!" }
			row.status.Text = labels[entryState] or entryState
			row.status.TextColor3 = protected and GREEN or RED
		end
		skipQuietPiece.holder.Visible = gameState:GetAttribute("QuietNight") == true and gameState:GetAttribute("SkipAvailable") == true
		skipNightPiece.holder.Visible = gameState:GetAttribute("QuietNight") ~= true
		if gameState:GetAttribute("QuietNight") and t >= creepyNext then
			creepyNext = t + 6 + math.random() * 7
			play(math.random() < 0.5 and "CreepyFootsteps" or "CreepyKnock", 0.8 + math.random() * 0.4)
		end
	end
	if revive.holder.Visible then
		reviveCountdown.Text = tostring(math.max(0, math.ceil(reviveDeadline - t)))
	end
	if arrow.Visible and arrowTarget then
		local center = centerOf(arrowTarget)
		local top = (arrowTarget.AbsolutePosition.Y - root.AbsolutePosition.Y) / uiScale.Scale
		arrow.Position = UDim2.fromOffset(center.X - 30, top - 62 + math.sin(t * 6) * 8)
	end
	local character = player.Character
	local hum = character and character:FindFirstChildOfClass("Humanoid")
	if hum then
		if t < shakeUntil then
			hum.CameraOffset = Vector3.new((math.random() - 0.5) * shakePower, (math.random() - 0.5) * shakePower, 0)
		elseif shakePower > 0 then
			shakePower = 0
			hum.CameraOffset = Vector3.new()
		end
	end
	for _, model in ipairs(customersFolder:GetChildren()) do animateModel(model, t, false) end
	for _, model in ipairs(monstersFolder:GetChildren()) do animateModel(model, t, true) end
	if monitor.Transparency < 1 and t - lastMonitor > 0.2 then
		lastMonitor = t
		local monsters = monstersFolder:GetChildren()
		for i, model in ipairs(monsters) do
			local dot = dots[i]
			if not dot then
				dot = new("Frame", { Size = UDim2.fromOffset(12, 12), AnchorPoint = Vector2.new(0.5, 0.5), BackgroundColor3 = Color3.fromRGB(255, 40, 40), BorderSizePixel = 0 }, monitorScreen)

				dots[i] = dot
			end

				dot.Position = UDim2.fromScale(math.clamp((rootPart.Position.X + 45) / 90, 0, 1), math.clamp((rootPart.Position.Z - 20) / 75, 0, 1))
			else
				dot.Visible = false
			end
		end
		for i = #monsters + 1, #dots do dots[i].Visible = false end
	end
end)

end

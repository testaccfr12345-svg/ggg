local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local ScriptContext = game:GetService("ScriptContext")
local player = Players.LocalPlayer

local gui = Instance.new("ScreenGui")
gui.Name = "IceCreamChecker"
gui.ResetOnSpawn = false
gui.DisplayOrder = 100
gui.Parent = player:WaitForChild("PlayerGui")

local box = Instance.new("TextLabel")
box.Size = UDim2.new(0.6, 0, 0.5, 0)
box.Position = UDim2.new(0.02, 0, 0.25, 0)
box.BackgroundColor3 = Color3.new(0, 0, 0)
box.BackgroundTransparency = 0.2
box.TextColor3 = Color3.new(1, 1, 1)
box.TextXAlignment = Enum.TextXAlignment.Left
box.TextYAlignment = Enum.TextYAlignment.Top
box.TextWrapped = true
box.TextScaled = true
box.Font = Enum.Font.Code
box.Text = "Checking..."
box.Parent = gui
local limit = Instance.new("UITextSizeConstraint")
limit.MaxTextSize = 18
limit.Parent = box

local close = Instance.new("TextButton")
close.Size = UDim2.new(0, 60, 0, 30)
close.Position = UDim2.new(1, -62, 0, 2)
close.Text = "HIDE"
close.BackgroundColor3 = Color3.fromRGB(200, 50, 50)
close.TextColor3 = Color3.new(1, 1, 1)
close.Parent = box
close.Activated:Connect(function() gui.Enabled = false end)

local clientError = "none"
ScriptContext.Error:Connect(function(message, trace)
	clientError = tostring(message) .. "\n" .. tostring(trace)
end)

local function describe(instance)
	if not instance then return "MISSING" end
	return "found (" .. instance.ClassName .. ")"
end

while true do
	local config = ReplicatedStorage:FindFirstChild("Config")
	local lines = {
		"ICE CREAM CHECKER",
		"Config in ReplicatedStorage: " .. describe(config),
		"Server status: " .. tostring(ReplicatedStorage:GetAttribute("IceCreamServer") or "NOT STARTED (Server2 is not running)"),
		"Server error: " .. tostring(ReplicatedStorage:GetAttribute("IceCreamServerError") or "none"),
		"Remotes: " .. describe(ReplicatedStorage:FindFirstChild("IceCreamRemotes")),
		"Map: " .. describe(workspace:FindFirstChild("IceCreamMap")),
		"Client UI: " .. describe(player.PlayerGui:FindFirstChild("IceCreamHUD")),
		"Client error: " .. clientError,
	}
	box.Text = table.concat(lines, "\n")
	task.wait(1)
end

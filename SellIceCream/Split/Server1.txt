local profiles = {}
local passCache = {}

local function hasPass(player, key)
	local info = Config.GamePasses[key]
	if not info or info.id == 0 then return false end
	passCache[player] = passCache[player] or {}
	if passCache[player][key] == nil then
		local ok, owns = pcall(function()
			return MarketplaceService:UserOwnsGamePassAsync(player.UserId, info.id)
		end)
		passCache[player][key] = ok and owns == true
	end
	return passCache[player][key]
end

local function anyoneHasPass(key)
	for _, player in ipairs(Players:GetPlayers()) do
		if profiles[player] and hasPass(player, key) then return player end
	end
	return nil
end

local function shopHasUpgrade(id)
	for _, profile in pairs(profiles) do
		if profile.data.upgrades[id] then return true end
	end
	return false
end

local function shopFlavors()
	local list = {}
	for _, flavor in ipairs(Config.Flavors) do
		for _, profile in pairs(profiles) do
			if profile.data.flavors[flavor.id] then
				table.insert(list, flavor.id)
				break
			end
		end
	end
	if #list == 0 then
		for _, id in ipairs(Config.StartFlavors) do table.insert(list, id) end
	end
	return list
end

local function cashMultiplier(player)
	local profile = profiles[player]
	if not profile then return 1 end
	local multiplier = 1 + profile.data.franchises * Config.FranchiseMultiplier
	if hasPass(player, "DoubleCash") then multiplier = multiplier * 2 end
	if hasPass(player, "VIP") then multiplier = multiplier * (1 + Config.VIPBonus) end
	return multiplier
end

local function awardBadge(player, key)
	local id = Config.Badges[key]
	if not id or id == 0 then return end
	task.spawn(function()
		pcall(function() BadgeService:AwardBadge(player.UserId, id) end)
	end)
end

local function joinList(set)
	local list = {}
	for id, owned in pairs(set) do
		if owned then table.insert(list, id) end
	end
	table.sort(list)
	return table.concat(list, ",")
end

local function today()
	return math.floor(os.time() / 86400)
end

local function dailyInfo(data)
	local day = today()
	if data.lastDaily == day then return false, 0 end
	local streak = 1
	if data.lastDaily == day - 1 then
		streak = data.dailyStreak + 1
		if streak > Config.DailyRewardMaxDays then streak = 1 end
	end
	return true, streak
end

local function publish(player)
	local profile = profiles[player]
	if not profile then return end
	local data = profile.data
	if profile.leaderCash then profile.leaderCash.Value = math.floor(data.cash) end
	if profile.leaderNight then profile.leaderNight.Value = data.streak end
	player:SetAttribute("Cash", math.floor(data.cash))
	player:SetAttribute("OwnedFlavors", joinList(data.flavors))
	player:SetAttribute("OwnedUpgrades", joinList(data.upgrades))
	player:SetAttribute("Streak", data.streak)
	player:SetAttribute("BestNight", data.bestNight)
	player:SetAttribute("Franchises", data.franchises)
	player:SetAttribute("Detectors", data.detectors)
	player:SetAttribute("FranchiseReady", data.cash >= Config.FranchiseCost)
	local ready, streak = dailyInfo(data)
	player:SetAttribute("DailyReady", ready)
	player:SetAttribute("DailyAmount", Config.DailyRewardBase * streak)
	player:SetAttribute("VIP", hasPass(player, "VIP"))
	player:SetAttribute("Multiplier", cashMultiplier(player))
end

local function addCash(player, amount, reason)
	local profile = profiles[player]
	if not profile or amount == 0 then return end
	profile.data.cash = math.max(0, profile.data.cash + amount)
	if amount > 0 then profile.today.earned = profile.today.earned + amount end
	publish(player)
	send(player, "Cash", math.floor(amount), reason)
end

local function sanitize(raw)
	local data = defaultData()
	if type(raw) ~= "table" then return data end
	data.cash = tonumber(raw.cash) or data.cash
	data.streak = tonumber(raw.streak) or 0
	data.bestNight = tonumber(raw.bestNight) or 0
	data.fakesCaught = tonumber(raw.fakesCaught) or 0
	data.franchises = tonumber(raw.franchises) or 0
	data.dailyStreak = tonumber(raw.dailyStreak) or 0
	data.lastDaily = tonumber(raw.lastDaily) or -1
	data.tutorialDone = raw.tutorialDone == true
	data.day = tonumber(raw.day) or 1
	data.highestDay = tonumber(raw.highestDay) or data.day
	data.detectors = tonumber(raw.detectors) or 0
	if type(raw.flavors) == "table" then
		for _, id in ipairs(raw.flavors) do
			if FLAVOR_BY_ID[id] then data.flavors[id] = true end
		end
	end
	if type(raw.upgrades) == "table" then
		for _, id in ipairs(raw.upgrades) do
			if UPGRADE_BY_ID[id] then data.upgrades[id] = true end
		end
	end
	if type(raw.receipts) == "table" then
		for _, id in ipairs(raw.receipts) do
			if type(id) == "string" then table.insert(data.receipts, id) end
		end
	end
	return data
end

local function serialize(data)
	local flavors, upgrades = {}, {}
	for id, owned in pairs(data.flavors) do if owned then table.insert(flavors, id) end end
	for id, owned in pairs(data.upgrades) do if owned then table.insert(upgrades, id) end end
	local receipts = {}
	for i = math.max(1, #data.receipts - 49), #data.receipts do table.insert(receipts, data.receipts[i]) end
	return {
		cash = math.floor(data.cash), flavors = flavors, upgrades = upgrades, streak = data.streak,
		bestNight = data.bestNight, fakesCaught = data.fakesCaught, franchises = data.franchises,
		dailyStreak = data.dailyStreak, lastDaily = data.lastDaily, tutorialDone = data.tutorialDone,
		day = data.day, highestDay = data.highestDay, detectors = data.detectors, receipts = receipts,
	}
end

local function withRetries(callback)
	for attempt = 1, 3 do
		local ok, result = pcall(callback)
		if ok then return true, result end
		task.wait(attempt * 1.5)
	end
	return false, nil
end

local function loadData(player)
	if not store then return IS_STUDIO, nil end
	local ok, raw = withRetries(function()
		return store:GetAsync("player_" .. player.UserId)
	end)
	return ok, raw
end

local function saveData(player)
	local profile = profiles[player]
	if not profile or not profile.canSave or not store then return false end
	local payload = serialize(profile.data)
	local ok = withRetries(function()
		store:SetAsync("player_" .. player.UserId, payload)
	end)
	if bestStore then
		pcall(function() bestStore:SetAsync(tostring(player.UserId), profile.data.bestNight) end)
	end
	return ok
end

local atmosphere = Lighting:FindFirstChildOfClass("Atmosphere") or make("Atmosphere", {}, Lighting)
local grade = Lighting:FindFirstChild("IceCreamGrade") or make("ColorCorrectionEffect", { Name = "IceCreamGrade" }, Lighting)

local function applyLighting(night)
	local info = TweenInfo.new(2.5, Enum.EasingStyle.Sine)
	if night then
		TweenService:Create(Lighting, info, {
			ClockTime = 0, Brightness = 0.4, Ambient = Color3.fromRGB(22, 20, 32), OutdoorAmbient = Color3.fromRGB(28, 26, 42),
			FogEnd = 110, FogStart = 10, FogColor = Color3.fromRGB(12, 10, 20),
		}):Play()
		TweenService:Create(atmosphere, info, { Density = 0.55, Color = Color3.fromRGB(25, 20, 35), Decay = Color3.fromRGB(40, 10, 20), Haze = 2.5 }):Play()
		TweenService:Create(grade, info, { TintColor = Color3.fromRGB(255, 205, 205), Saturation = -0.25, Contrast = 0.15, Brightness = -0.05 }):Play()
	else
		TweenService:Create(Lighting, info, {
			ClockTime = 14, Brightness = 2.4, Ambient = Color3.fromRGB(150, 140, 160), OutdoorAmbient = Color3.fromRGB(175, 165, 190),
			FogEnd = 1000, FogStart = 200, FogColor = Color3.fromRGB(255, 220, 235),
		}):Play()
		TweenService:Create(atmosphere, info, { Density = 0.25, Color = Color3.fromRGB(255, 225, 238), Decay = Color3.fromRGB(255, 200, 220), Haze = 1 }):Play()
		TweenService:Create(grade, info, { TintColor = Color3.fromRGB(255, 250, 252), Saturation = 0.15, Contrast = 0.05, Brightness = 0.02 }):Play()
	end
end

local function updateShopVisuals()
	local flavors = shopFlavors()
	local owned = {}
	for _, id in ipairs(flavors) do owned[id] = true end
	for id, scoop in pairs(tubParts) do
		scoop.Transparency = owned[id] and 0 or 1
	end
	gameState:SetAttribute("ShopFlavors", table.concat(flavors, ","))
	local upgrades = {}
	for _, upgrade in ipairs(Config.Upgrades) do
		if shopHasUpgrade(upgrade.id) then table.insert(upgrades, upgrade.id) end
	end
	gameState:SetAttribute("ShopUpgrades", table.concat(upgrades, ","))
	local camera = shopHasUpgrade("SecurityCamera")
	monitor.Transparency = camera and 0 or 1
	monitorGui.Enabled = camera
	shopSignTitle.TextColor3 = shopHasUpgrade("NeonSign") and Color3.fromRGB(255, 60, 200) or Color3.fromRGB(255, 120, 170)
	shopSign.Material = shopHasUpgrade("NeonSign") and Enum.Material.Neon or Enum.Material.SmoothPlastic
	local names = {}
	local franchise = 0
	for player, profile in pairs(profiles) do
		table.insert(names, player.DisplayName)
		franchise = math.max(franchise, profile.data.franchises)
	end
	if #names == 0 then
		nameSign.Text = "OPEN SOON"
	else
		local text = table.concat(names, " & ") .. "'S SHOP"
		if franchise > 0 then text = text .. "  #" .. (franchise + 1) end
		nameSign.Text = text
	end
	local maxHealth = shopHasUpgrade("ReinforcedDoor") and Config.ReinforcedDoorHealth or Config.DoorHealth
	entries.Front.maxHealth = maxHealth
	local autoOwner = anyoneHasPass("AutoWorker")
	if autoOwner and not worker then
		task.spawn(function()
			local desc = Instance.new("HumanoidDescription")
			desc.HeadColor = Color3.fromRGB(255, 220, 190)
			desc.LeftArmColor = desc.HeadColor
			desc.RightArmColor = desc.HeadColor
			desc.TorsoColor = PINK
			desc.LeftLegColor = WHITE
			desc.RightLegColor = WHITE
			local ok, model = pcall(function()
				return Players:CreateHumanoidModelFromDescription(desc, Enum.HumanoidRigType.R15)
			end)
			if ok and model and not worker then
				model.Name = "AutoWorker"
				model:PivotTo(S(6, 4, 0))
				for _, p in ipairs(model:GetDescendants()) do
					if p:IsA("BasePart") then p.Anchored = true end
				end
				local hum = model:FindFirstChildOfClass("Humanoid")
				if hum then hum.DisplayName = "Auto-Worker" end
				model.Parent = shopFolder
				worker = model
			end
		end)
	elseif not autoOwner and worker then
		worker:Destroy()
		worker = nil
	end
end

refreshShop = function()
	updateShopVisuals()
	for player in pairs(profiles) do publish(player) end
end

layout.SortOrder = Enum.SortOrder.LayoutOrder

toggle.Activated:Connect(function()
	panel.Visible = not panel.Visible
end)

local count = 0
local COLORS = {
	[Enum.MessageType.MessageOutput] = Color3.new(1, 1, 1),
	[Enum.MessageType.MessageInfo] = Color3.fromRGB(120, 200, 255),
	[Enum.MessageType.MessageWarning] = Color3.fromRGB(255, 200, 60),
	[Enum.MessageType.MessageError] = Color3.fromRGB(255, 80, 80),
}

local function addLine(text, color)
	count = count + 1
	local line = Instance.new("TextLabel")
	line.Size = UDim2.new(1, -10, 0, 0)
	line.AutomaticSize = Enum.AutomaticSize.Y
	line.BackgroundTransparency = 1
	line.TextColor3 = color or Color3.new(1, 1, 1)
	line.TextXAlignment = Enum.TextXAlignment.Left
	line.TextWrapped = true
	line.Font = Enum.Font.Code
	line.TextSize = 14
	line.Text = text
	line.LayoutOrder = count
	line.Parent = list
	local lines = {}
	for _, child in ipairs(list:GetChildren()) do
		if child:IsA("TextLabel") then table.insert(lines, child) end
	end
	if #lines > 200 then
		table.sort(lines, function(a, b) return a.LayoutOrder < b.LayoutOrder end)
		lines[1]:Destroy()
	end
	list.CanvasPosition = Vector2.new(0, 1000000)
end

for _, entry in ipairs(LogService:GetLogHistory()) do
	addLine(entry.message, COLORS[entry.messageType])
end
LogService.MessageOut:Connect(function(message, messageType)
	addLine(message, COLORS[messageType])
end)

local lastServerError = nil

local function describe(instance)
	if not instance then return "MISSING" end
	return "found (" .. instance.ClassName .. ")"
end

while true do
	local serverError = ReplicatedStorage:GetAttribute("IceCreamServerError")
	if serverError and serverError ~= lastServerError then
		lastServerError = serverError
		addLine("[SERVER ERROR] " .. tostring(serverError), Color3.fromRGB(255, 80, 80))
	end
	status.Text = table.concat({
		"Config (ReplicatedStorage): " .. describe(ReplicatedStorage:FindFirstChild("Config")),
		"Server: " .. tostring(ReplicatedStorage:GetAttribute("IceCreamServer") or "NOT RUNNING"),
		"Remotes: " .. describe(ReplicatedStorage:FindFirstChild("IceCreamRemotes")),
		"Map: " .. describe(workspace:FindFirstChild("IceCreamMap")),
		"Game UI: " .. describe(player.PlayerGui:FindFirstChild("IceCreamHUD")),
	}, "\n")
	task.wait(1)
end

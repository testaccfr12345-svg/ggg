local function showBubble(customer)
	if customer.bubble or not customer.model.Parent then return end
	local head = customer.model:FindFirstChild("Head")
	if not head then return end
	local gui = make("BillboardGui", {
		Name = "OrderBubble",
		Size = UDim2.fromOffset(230, 96),
		StudsOffset = Vector3.new(0, 3.2, 0),
		AlwaysOnTop = true,
		MaxDistance = 70,
		LightInfluence = 0,
	}, head)
	local bubble = make("Frame", { Size = UDim2.new(1, 0, 0, 62), BackgroundColor3 = WHITE, BorderSizePixel = 0 }, gui)
	make("UICorner", { CornerRadius = UDim.new(0, 16) }, bubble)
	make("UIStroke", { Color = DARK, Thickness = 3 }, bubble)
	local label = make("TextLabel", {
		Size = UDim2.new(1, -14, 1, -8),
		Position = UDim2.fromOffset(7, 4),
		BackgroundTransparency = 1,
		Text = customer.display,
		TextScaled = true,
		TextWrapped = true,
		Font = Enum.Font.FredokaOne,
		TextColor3 = DARK,
	}, bubble)
	local back = make("Frame", { Size = UDim2.new(0.8, 0, 0, 14), Position = UDim2.new(0.1, 0, 0, 72), BackgroundColor3 = DARK, BorderSizePixel = 0 }, gui)
	make("UICorner", { CornerRadius = UDim.new(1, 0) }, back)
	local bar = make("Frame", { Size = UDim2.new(1, -4, 1, -4), Position = UDim2.fromOffset(2, 2), BackgroundColor3 = Color3.fromRGB(110, 230, 110), BorderSizePixel = 0 }, back)
	make("UICorner", { CornerRadius = UDim.new(1, 0) }, bar)
	customer.bubble = gui
	customer.bubbleLabel = label
	customer.patienceBar = bar
	customer.patienceBack = back
	back.Visible = false
end

local function currentCustomer()
	local first = state.line[1]
	if first and first.atCounter and not first.leaving then return first end
	return nil
end

local function updateOrderScreen()
	local c = currentCustomer()
	if state.phase ~= "Day" then
		orderScreenText.Text = state.phase == "Night" and "CLOSED" or "..."
	elseif c then
		orderScreenText.Text = c.display
	else
		orderScreenText.Text = "NEXT!"
	end
	gameState:SetAttribute("CurrentOrder", c and c.display or "")
	gameState:SetAttribute("CurrentCustomerId", c and c.id or 0)
end

local customerId = 0

local function removeFromLine(customer)
	for i, other in ipairs(state.line) do
		if other == customer then
			table.remove(state.line, i)
			break
		end
	end
	updateOrderScreen()
end

local function customerLeave(customer, speed, mood)
	if customer.leaving then return end
	customer.leaving = true
	customer.atCounter = false
	removeFromLine(customer)
	if customer.patienceBack then customer.patienceBack.Visible = false end
	if mood then setFace(customer.model, mood) end
	task.spawn(function()
		local hum = customer.model:FindFirstChildOfClass("Humanoid")
		if hum then hum.WalkSpeed = speed or 12 end
		task.wait(0.6)
		if customer.bubble then customer.bubble:Destroy() end
		walkTo(customer.model, SP(0, 1, -19), 8)
		walkTo(customer.model, STREET_FRONT, 8)
		walkTo(customer.model, STREET_SPAWN, 25)
		customer.model:Destroy()
		state.customers[customer.id] = nil
	end)
end

local tutorialPlayers = {}

local function tutorialStep(step)
	for player in pairs(tutorialPlayers) do
		if player.Parent then send(player, "Tutorial", step) end
	end
end

local function spawnCustomer()
	state.spawnedToday = state.spawnedToday + 1
	local forcedFake = state.tutorial and state.spawnedToday == Config.TutorialFakeNumber
	local fake = forcedFake or (not state.tutorial or state.spawnedToday > Config.TutorialFakeNumber) and rng:NextNumber() < fakeChance(state.day)
	if state.tutorial and state.spawnedToday < Config.TutorialFakeNumber then fake = false end
	local tell = nil
	if fake then tell = forcedFake and "BlackEyes" or pickTell(state.day) end
	local model = buildRig("customer")
	if not model then return end
	customerId = customerId + 1
	local items, display = makeOrder(state.day, tell)
	local customer = {
		id = customerId, model = model, fake = fake, tell = tell, items = items, display = display,
		atCounter = false, leaving = false, patience = Config.Patience, tutorialFake = forcedFake,
	}
	model.Name = "Customer"
	model:SetAttribute("CustomerId", customer.id)
	local hum = model:FindFirstChildOfClass("Humanoid")
	hum.WalkSpeed = 12
	if tell == "BlackEyes" then setFace(model, "black") end
	if tell == "Float" then hum.HipHeight = hum.HipHeight + Config.FloatHeight end
	if tell == "Still" then model:SetAttribute("Still", true) end
	if tell == "HeadFollow" then model:SetAttribute("HeadFollow", true) end
	if tell == "NoShadow" then
		for _, p in ipairs(model:GetDescendants()) do
			if p:IsA("BasePart") then p.CastShadow = false end
		end
	end
	if tell == "LongArms" then scaleLimbs(model, ARM_PARTS, Config.LongArmScale) end
	placeRig(model, customerFolder, CFrame.new(STREET_SPAWN + Vector3.new(0, 3, 0)))
	state.customers[customer.id] = customer
	table.insert(state.line, customer)
	if tell == "EarlyBubble" then showBubble(customer) end
	if tell == "Flicker" then
		task.spawn(function()
			local skin = model:GetAttribute("Skin")
			while model.Parent and not customer.leaving do
				task.wait(rng:NextNumber(4, 6))
				if not model.Parent then break end
				local limbs = {}
				for _, name in ipairs({ "Head", "LeftUpperArm", "LeftLowerArm", "LeftHand", "RightUpperArm", "RightLowerArm", "RightHand" }) do
					local limb = model:FindFirstChild(name)
					if limb then
						limb.Color = Color3.fromRGB(130, 130, 135)
						table.insert(limbs, limb)
					end
				end
				task.wait(0.2)
				for _, limb in ipairs(limbs) do
					if limb.Parent then limb.Color = skin end
				end
			end
		end)
	end
	task.spawn(function()
		walkTo(model, STREET_FRONT, 30)
		while model.Parent and not customer.leaving do
			local index = 0
			for i, other in ipairs(state.line) do
				if other == customer then index = i break end
			end
			if index == 0 then break end
			local target = LINE_SLOTS[math.min(index, #LINE_SLOTS)]
			local root = model:FindFirstChild("HumanoidRootPart")
			if root and (Vector3.new(target.X, root.Position.Y, target.Z) - root.Position).Magnitude > 1.6 then
				walkTo(model, target, 3)
			elseif index == 1 and not customer.atCounter then
				customer.atCounter = true
				customer.arrivedAt = os.clock()
				if root then model:PivotTo(CFrame.lookAt(root.Position, root.Position + Vector3.new(0, 0, 1))) end
				showBubble(customer)
				if customer.bubbleLabel then customer.bubbleLabel.Text = customer.display end
				if customer.patienceBack then customer.patienceBack.Visible = true end
				updateOrderScreen()
				if customer.tutorialFake then tutorialStep("refuse") end
				if customer.tell == "SaysName" then
					task.delay(1.5, function()
						local players = Players:GetPlayers()
						if #players > 0 and customer.bubbleLabel and not customer.leaving then
							customer.bubbleLabel.Text = customer.display .. "\n..." .. randomOf(players).Name .. "..."
						end
					end)
				end
			end
			task.wait(0.15)
		end
	end)
end

local serveCooldown = {}

local function finishCustomer(player, customer, served)
	local profile = profiles[player]
	if customer.fake then
		if served then
			state.fakesServedToday = state.fakesServedToday + 1
			if profile then profile.today.missed = profile.today.missed + 1 end
			setFace(customer.model, "creepy")
			sendAll("Sound", "FakeServed")
			customerLeave(customer, 6, nil)
		end
	end
end

local function serveCustomer(player, customer, orderMap, auto)
	local profile = profiles[player]
	if not profile then return end
	if customer.fake then
		finishCustomer(player, customer, true)
		if profile then profile.today.served = profile.today.served + 1 end
		return
	end
	local correct = true
	local expected = {}
	for _, item in ipairs(customer.items) do expected[item.flavor] = (expected[item.flavor] or 0) + item.scoops end
	if not auto then
		for flavor, count in pairs(expected) do
			if orderMap[flavor] ~= count then correct = false end
		end
		for flavor, count in pairs(orderMap) do
			if expected[flavor] ~= count then correct = false end
		end
	end
	if not correct then
		toast(player, "Wrong order!", "bad")
		customerLeave(customer, 14, "angry")
		return
	end
	local base = 0
	for flavor, count in pairs(expected) do base = base + FLAVOR_BY_ID[flavor].price * count end
	local tip = 0
	if os.clock() - (customer.arrivedAt or os.clock()) <= Config.TipTime then
		local rate = Config.TipBonus
		if shopHasUpgrade("TipJar") then rate = rate + Config.TipJarBonus end
		tip = base * rate
	end
	local total = math.floor((base + tip) * cashMultiplier(player) + 0.5)
	addCash(player, total, tip > 0 and "Tip!" or nil)
	profile.today.served = profile.today.served + 1
	customerLeave(customer, 12, "happy")
end

ServeRemote.OnServerEvent:Connect(function(player, orderMap)
	local profile = profiles[player]
	if not profile or state.phase ~= "Day" or type(orderMap) ~= "table" then return end
	local now = os.clock()
	local cooldown = shopHasUpgrade("FastScoop") and Config.FastServeTime or Config.ServeTime
	if serveCooldown[player] and now - serveCooldown[player] < cooldown * 0.8 then return end
	serveCooldown[player] = now
	local clean = {}
	local total = 0
	local owned = {}
	for _, id in ipairs(shopFlavors()) do owned[id] = true end
	for flavor, count in pairs(orderMap) do
		if type(flavor) ~= "string" or not owned[flavor] or type(count) ~= "number" or count ~= math.floor(count) or count < 1 or count > Config.MaxScoopsPerFlavor then
			return
		end
		clean[flavor] = count
		total = total + count
	end
	if total < 1 or total > Config.MaxScoopsTotal then return end
	local customer = currentCustomer()
	if not customer then
		toast(player, "No customer at the counter!", "bad")
		return
	end
	local character = player.Character
	local root = character and character:FindFirstChild("HumanoidRootPart")
	if not root or (root.Position - SP(0, 3, 0)).Magnitude > 40 then
		toast(player, "Get behind your counter!", "bad")
		return
	end
	serveCustomer(player, customer, clean, false)
end)

RefuseRemote.OnServerEvent:Connect(function(player)
	local profile = profiles[player]
	if not profile or state.phase ~= "Day" then return end
	local now = os.clock()
	if serveCooldown[player] and now - serveCooldown[player] < 0.4 then return end
	serveCooldown[player] = now
	local customer = currentCustomer()
	if not customer then return end
	if customer.fake then
		profile.data.fakesCaught = profile.data.fakesCaught + 1
		profile.today.caught = profile.today.caught + 1
		state.fakesCaughtToday = state.fakesCaughtToday + 1
		if profile.data.fakesCaught >= 10 then awardBadge(player, "Catch10") end
		if profile.data.fakesCaught >= 100 then awardBadge(player, "Catch100") end
		addCash(player, math.floor(Config.RefuseFakeBonus * cashMultiplier(player)), "Fake caught!")
		local root = customer.model:FindFirstChild("HumanoidRootPart")
		if root then soundIn(root, "Hiss", 1) end
		sendAll("Hiss", customer.id)
		task.spawn(function()
			local head = customer.model:FindFirstChild("Head")
			local gui = head and head:FindFirstChild("FaceGui")
			for _ = 1, 10 do
				if gui then
					for _, frame in ipairs(gui:GetChildren()) do
						if frame:IsA("Frame") then
							frame.Position = frame.Position + UDim2.fromScale(rng:NextNumber(-0.08, 0.08), rng:NextNumber(-0.08, 0.08))
							frame.BackgroundColor3 = rng:NextNumber() < 0.5 and Color3.new(0, 0, 0) or RED
						end
					end
				end
				task.wait(0.05)
			end
		end)
		customerLeave(customer, 30, nil)
	else
		profile.data.cash = math.max(0, profile.data.cash - Config.RefuseRealPenalty)
		publish(player)
		send(player, "Cash", -Config.RefuseRealPenalty, "Complaint!")
		customerLeave(customer, 10, "sad")
	end
	publish(player)
end)

		character:PivotTo(cf)
	end
end

local function playersInGame()
	local list = {}
	for player in pairs(profiles) do
		if player.Parent then table.insert(list, player) end
	end
	return list
end

local function runDay()
	setPhase("Day", Config.DayLength)
	applyLighting(false)
	setLights(true)
	state.line = {}
	state.spawnedToday = 0
	state.fakesServedToday = 0
	state.fakesCaughtToday = 0
	tutorialPlayers = {}
	state.tutorial = false
	for player, profile in pairs(profiles) do
		profile.today = { earned = 0, served = 0, caught = 0, missed = 0 }
		profile.failed = false
		profile.caught = false
		if not profile.data.tutorialDone then
			tutorialPlayers[player] = true
			state.tutorial = true
		end
	end
	state.tutorialNight = state.tutorial
	for _, entry in pairs(entries) do
		entry.state = entry.startState
		if entry.name == "Front" then entry.state = "Open" end
	end
	resetEntries()
	for _, player in ipairs(playersInGame()) do
		teleport(player, COUNTER_SPOT)
		local hum = player.Character and player.Character:FindFirstChildOfClass("Humanoid")
		if hum then hum.WalkSpeed = Config.PlayerSpeed end
	end
	updateOrderScreen()
	sendAll("PhaseStart", "Day", state.day)
	if state.tutorial then
		task.delay(1.5, function() tutorialStep("flavors") end)
	end
	local nextSpawn = os.clock() + 2
	local autoTimers = {}
	while workspace:GetServerTimeNow() < state.endsAt do
		if #playersInGame() == 0 then break end
		local now = os.clock()
		local interval = math.max(Config.SpawnIntervalMin, Config.SpawnIntervalStart - (state.day - 1) * Config.SpawnIntervalStep)
		if shopHasUpgrade("NeonSign") then interval = interval / (1 + Config.NeonSignBonus) end
		if now >= nextSpawn and #state.line < Config.MaxLine then
			nextSpawn = now + interval
			task.spawn(spawnCustomer)
		end
		for _, customer in pairs(state.customers) do
			if customer.atCounter and not customer.leaving then
				local left = Config.Patience - (now - customer.arrivedAt)
				if customer.patienceBar then
					local ratio = math.clamp(left / Config.Patience, 0, 1)
					customer.patienceBar.Size = UDim2.new(ratio, -4, 1, -4)
					customer.patienceBar.BackgroundColor3 = ratio > 0.5 and Color3.fromRGB(110, 230, 110) or (ratio > 0.25 and Color3.fromRGB(255, 210, 60) or Color3.fromRGB(240, 70, 70))
				end
				if left <= 0 then
					customerLeave(customer, 14, "angry")
				else
					local autoOwner = anyoneHasPass("AutoWorker")
					if autoOwner and now - customer.arrivedAt >= Config.AutoWorkerDelay and not autoTimers[customer.id] then
						autoTimers[customer.id] = true
						serveCustomer(autoOwner, customer, nil, true)
					end
				end
			end
		end
		task.wait(0.2)
	end
	for _, customer in pairs(state.customers) do
		customerLeave(customer, 16, nil)
	end
	task.wait(1)
end

local function setFrozen(player, frozen)
	local character = player.Character
	local hum = character and character:FindFirstChildOfClass("Humanoid")
	local root = character and character:FindFirstChild("HumanoidRootPart")
	if hum then
		hum.WalkSpeed = frozen and 0 or Config.PlayerSpeed
		hum.JumpPower = frozen and 0 or 50
		hum.JumpHeight = frozen and 0 or 7.2
	end
	if root then root.Anchored = frozen end
end

local function setFlashlight(player, on)
	local profile = profiles[player]
	if not profile then return end
	if on and (state.phase ~= "Night" or profile.battery <= 0 or profile.caught or profile.failed) then on = false end
	profile.flashOn = on
	player:SetAttribute("FlashOn", on)
	local head = player.Character and player.Character:FindFirstChild("Head")
	local existing = head and head:FindFirstChild("Flashlight")
	if on and head and not existing then
		local better = shopHasUpgrade("BetterFlashlight")
		make("SpotLight", { Name = "Flashlight", Range = Config.FlashlightRange + 5, Angle = better and 55 or 45, Brightness = better and 8 or 5, Face = Enum.NormalId.Front, Shadows = true, Color = Color3.fromRGB(255, 245, 220) }, head)
	elseif not on and existing then
		existing:Destroy()
	end
end

local function newMonster()
	local model = buildRig("monster")
	if not model then return nil end
	model.Name = "Monster"
	local spawnPoint = randomOf({ SP(0, 1, 32), SP(-30, 1, 30), SP(30, 1, 30) })
	placeRig(model, monsterFolder, CFrame.new(spawnPoint + Vector3.new(0, 4, 0)))
	local hum = model:FindFirstChildOfClass("Humanoid")
	hum.WalkSpeed = Config.MonsterSpeed
	local monster = { model = model, hum = hum, root = model:FindFirstChild("HumanoidRootPart"), state = "wander", exposure = {}, alive = true, tick = 0 }
	table.insert(state.monsters, monster)
	return monster
end

local function monsterTarget(monster)
	local best, bestDist = nil, math.huge
	for player, profile in pairs(profiles) do
		local root = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
		if root and not profile.caught and not profile.failed and inBox(root.Position, PLOT_MIN, PLOT_MAX) then
			local d = (root.Position - monster.root.Position).Magnitude
			if d < bestDist then best, bestDist = player, d end
		end
	end
	return best, bestDist
end

local function chooseEntry()
	local options = {}
	local total = 0
	for _, entry in pairs(entries) do
		local weight = 1
		if entry.state == "Open" or entry.state == "Broken" or entry.state == "Closed" then
			weight = 6
		elseif entry.state == "Glass" then
			weight = 4
		else
			weight = 1 + (1 - entry.health / entry.maxHealth) * 3
		end
		total = total + weight
		table.insert(options, { entry = entry, weight = weight })
	end
	local roll = rng:NextNumber() * total
	for _, option in ipairs(options) do
		roll = roll - option.weight
		if roll <= 0 then return option.entry end
	end
	return entries.Front
end

local alarmTimes = {}
local function alarm(entry)
	if not shopHasUpgrade("AlarmBell") then return end
	local now = os.clock()
	if alarmTimes[entry.name] and now - alarmTimes[entry.name] < 2.5 then return end
	alarmTimes[entry.name] = now
	sendAll("Alarm", entry.display)
end

local function breakEntry(entry)
	entry.state = "Broken"
	entry.health = 0
	setEntryVisual(entry)
	refreshPrompt(entry)
	sendAll("EntryBroken", entry.display)
end

local function retreatMonster(monster, seconds)
	monster.state = "retreat"
	monster.retreatUntil = os.clock() + seconds
	monster.exposure = {}
	if monster.root and inBox(monster.root.Position, SHOP_MIN, SHOP_MAX) then
		monster.model:PivotTo(CFrame.new(randomOf(OUTSIDE_POINTS) + Vector3.new(0, 4, 0)))
	end
	monster.hum:MoveTo(randomOf(OUTSIDE_POINTS))
end

local function flashMonster(monster, player)
	if monster.state == "retreat" then return end
	soundIn(monster.root, "MonsterScream", 1.2)
	sendAll("MonsterFlinch", monster.model)
	monster.hum.WalkSpeed = 0
	task.delay(0.5, function()
		if monster.alive then
			monster.hum.WalkSpeed = Config.MonsterSpeed
			retreatMonster(monster, Config.MonsterRetreatTime)
		end
	end)
	monster.state = "flinch"
	if player then toast(player, "It ran away!", "good") end
end

local function failPlayer(player)
	local profile = profiles[player]
	if not profile or not profile.caught then return end
	profile.caught = false
	profile.failed = true
	profile.data.cash = math.max(0, profile.data.cash - profile.today.earned)
	profile.today.lost = profile.today.earned
	profile.data.streak = 0
	publish(player)
	setFrozen(player, false)
	teleport(player, LOBBY_SPOT)
	send(player, "Failed")
	local active = 0
	for p, other in pairs(profiles) do
		if p.Parent and not other.failed then active = active + 1 end
	end
	if active == 0 then state.nightOver = true end
end

local function revivePlayer(player)
	local profile = profiles[player]
	if not profile or not profile.caught then return false end
	profile.caught = false
	profile.reviving = false
	for _, monster in ipairs(state.monsters) do
		if monster.alive then retreatMonster(monster, 6) end
	end
	setFrozen(player, false)
	teleport(player, COUNTER_SPOT)
	send(player, "Revived")
	return true
end

local function catchPlayer(player)
	local profile = profiles[player]
	if not profile or profile.caught or profile.failed then return end
	profile.caught = true
	setFlashlight(player, false)
	setFrozen(player, true)
	local deadline = os.clock() + Config.ReviveTime
	send(player, "Caught", Config.ReviveTime, Config.Products.Revive.id == 0 and IS_STUDIO)
	task.spawn(function()
		while profile.caught and player.Parent do
			if os.clock() > deadline and not profile.reviving then break end
			if profile.reviving and os.clock() > deadline + 25 then break end
			task.wait(0.2)
		end
		if profile.caught and player.Parent then failPlayer(player) end
	end)
end

local function monsterLoop(monster)
	local path = PathfindingService:CreatePath({ AgentRadius = 2, AgentHeight = 7, AgentCanJump = false })
	local lastPath = 0
	local lastBang = 0
	while monster.alive and state.phase == "Night" and not state.nightOver do
		local dt = 0.15
		monster.tick = monster.tick + 1
		local now = os.clock()
		if monster.state ~= "flinch" then
			monster.hum.WalkSpeed = (monster.tick % 3 == 0) and 0 or Config.MonsterSpeed * 1.45
		end
		if monster.state == "wander" then
			if not monster.wanderUntil then
				monster.wanderUntil = now + rng:NextNumber(Config.MonsterWanderMin, Config.MonsterWanderMax)
				monster.hum:MoveTo(randomOf(OUTSIDE_POINTS))
			elseif now >= monster.wanderUntil then
				monster.wanderUntil = nil
				monster.entry = chooseEntry()
				monster.state = "approach"
				monster.lastMove = 0
			end
		elseif monster.state == "approach" then
			local entry = monster.entry
			if now - (monster.lastMove or 0) > 1.5 then
				monster.hum:MoveTo(entry.outside)
				monster.lastMove = now
			end
			local flat = Vector3.new(entry.outside.X, monster.root.Position.Y, entry.outside.Z)
			if (monster.root.Position - flat).Magnitude < 3.2 then
				monster.state = "attack"
				monster.crawl = 0
			end
		elseif monster.state == "attack" then
			local entry = monster.entry
			if entry.state == "Closed" then
				entry.state = "Open"
				setEntryVisual(entry)
				refreshPrompt(entry)
			end
			if entry.state == "Open" or entry.state == "Broken" then
				monster.state = "enter"
			elseif entry.state == "Glass" then
				monster.crawl = (monster.crawl or 0) + dt
				if not monster.crawlSound then
					monster.crawlSound = true
					soundIn(entry.glass, "WindowCrawl", 1)
					sendAll("Crawl", entry.display)
				end
				alarm(entry)
				if monster.crawl >= Config.WindowCrawlTime then
					breakEntry(entry)
					monster.crawlSound = nil
					monster.state = "enter"
				end
			else
				entry.health = entry.health - Config.MonsterDamage * dt
				refreshEntryAttributes(entry)
				refreshPrompt(entry)
				alarm(entry)
				if now - lastBang > 0.55 then
					lastBang = now
					soundIn(entry.kind == "door" and entry.door or entry.boards[2], "DoorBang", 1.2)
					sendAll("Bang", entry.display, entry.kind == "door" and entry.door.Position or entry.boards[2].Position)
				end
				if entry.health <= 0 then
					breakEntry(entry)
					monster.state = "enter"
				end
			end
		elseif monster.state == "enter" then
			monster.model:PivotTo(CFrame.new(monster.entry.inside + Vector3.new(0, 4, 0)))
			monster.state = "chase"
			lastPath = 0
		elseif monster.state == "chase" then
			local target, distance = monsterTarget(monster)
			if target then
				local targetRoot = target.Character.HumanoidRootPart
				if distance <= Config.CatchDistance then
					catchPlayer(target)
				elseif now - lastPath > 0.6 then
					lastPath = now
					local ok = pcall(function() path:ComputeAsync(monster.root.Position, targetRoot.Position) end)
					local moved = false
					if ok and path.Status == Enum.PathStatus.Success then
						local waypoints = path:GetWaypoints()
						local waypoint = waypoints[math.min(3, #waypoints)]
						if waypoint then
							monster.hum:MoveTo(waypoint.Position)
							moved = true
						end
					end
					if not moved then monster.hum:MoveTo(targetRoot.Position) end
				end
			else
				if now - lastPath > 1.5 then
					lastPath = now
					monster.hum:MoveTo(SP(rng:NextNumber(-14, 14), 1, rng:NextNumber(-12, 10)))
				end
			end
		elseif monster.state == "retreat" then
			if now >= (monster.retreatUntil or 0) then
				monster.entry = chooseEntry()
				monster.state = "approach"
				monster.lastMove = 0
			end
		end
		task.wait(dt)
	end
end

local function clearMonsters()
	for _, monster in ipairs(state.monsters) do
		monster.alive = false
		local model = monster.model
		for _, p in ipairs(model:GetDescendants()) do
			if p:IsA("BasePart") then
				TweenService:Create(p, TweenInfo.new(1.2), { Transparency = 1 }):Play()
			end
		end
		task.delay(1.3, function() if model.Parent then model:Destroy() end end)
	end
	state.monsters = {}
end

local function flashlightTick(dt)
	for player, profile in pairs(profiles) do
		local maxBattery = shopHasUpgrade("BetterFlashlight") and Config.FlashlightBatteryUpgraded or Config.FlashlightBattery
		if profile.flashOn then
			profile.battery = math.max(0, profile.battery - dt)
			if profile.battery <= 0 then setFlashlight(player, false) end
		else
			profile.battery = math.min(maxBattery, profile.battery + Config.FlashlightRecharge * dt)
		end
		player:SetAttribute("Battery", math.floor(profile.battery * 10) / 10)
		player:SetAttribute("BatteryMax", maxBattery)
		if profile.flashOn and state.phase == "Night" then
			local character = player.Character
			local head = character and character:FindFirstChild("Head")
			local root = character and character:FindFirstChild("HumanoidRootPart")
			if head and root then
				local stun = shopHasUpgrade("BetterFlashlight") and Config.FlashlightStunUpgraded or Config.FlashlightStun
				for _, monster in ipairs(state.monsters) do
					if monster.alive and monster.state ~= "retreat" and monster.state ~= "flinch" and monster.root then
						local offset = monster.root.Position - head.Position
						local lit = false
						if offset.Magnitude <= Config.FlashlightRange and offset.Magnitude > 0 then
							local dot = root.CFrame.LookVector:Dot(offset.Unit)
							if dot > 0.72 then
								local params = RaycastParams.new()
								params.FilterType = Enum.RaycastFilterType.Exclude
								params.FilterDescendantsInstances = { character, monster.model, customerFolder }
								local hit = workspace:Raycast(head.Position, offset, params)
								lit = hit == nil or (hit.Instance and hit.Instance.Transparency > 0.5)
							end
						end
						if lit then
							monster.exposure[player] = (monster.exposure[player] or 0) + dt
							if monster.exposure[player] >= stun then flashMonster(monster, player) end
						else
							monster.exposure[player] = 0
						end
					end
				end
			end
		end
	end
end

local function runNight()
	state.nightOver = false
	state.survivedBySkip = false
	state.monsters = {}
	setPhase("Night", Config.NightLength)
	applyLighting(true)
	setLights(false)
	updateOrderScreen()
	resetEntries()
	for _, entry in pairs(entries) do refreshPrompt(entry) end
	local count = state.fakesServedToday
	if state.tutorialNight then count = math.min(count, 1) end
	state.quiet = count == 0
	state.skipAvailable = false
	gameState:SetAttribute("QuietNight", state.quiet)
	gameState:SetAttribute("SkipAvailable", false)
	for player, profile in pairs(profiles) do
		local maxBattery = shopHasUpgrade("BetterFlashlight") and Config.FlashlightBatteryUpgraded or Config.FlashlightBattery
		profile.battery = maxBattery
		profile.caught = false
		profile.failed = false
		teleport(player, COUNTER_SPOT)
	end
	sendAll("PhaseStart", "Night", state.day)
	if state.tutorialNight then tutorialStep("night") end
	local spawned = 0
	local nextSpawn = os.clock() + Config.MonsterFirstSpawn
	local started = os.clock()
	local last = os.clock()
	while workspace:GetServerTimeNow() < state.endsAt and not state.nightOver do
		if #playersInGame() == 0 then break end
		local now = os.clock()
		local dt = now - last
		last = now
		if spawned < count and now >= nextSpawn then
			spawned = spawned + 1
			nextSpawn = now + Config.MonsterSpawnInterval
			task.spawn(function()
				local monster = newMonster()
				if monster then monsterLoop(monster) end
			end)
		end
		if state.quiet and not state.skipAvailable and now - started >= Config.SkipQuietAfter then
			state.skipAvailable = true
			gameState:SetAttribute("SkipAvailable", true)
		end
		for player, entry in pairs(repairing) do
			if player.Parent and entry.health < entry.maxHealth and (entry.state == "Locked" or entry.state == "Boarded") then
				entry.health = math.min(entry.maxHealth, entry.health + Config.RepairPerSecond * dt)
				refreshEntryAttributes(entry)
				if entry.health >= entry.maxHealth then
					repairing[player] = nil
					refreshPrompt(entry)
				end
			else
				repairing[player] = nil
			end
		end
		flashlightTick(dt)
		task.wait(0.1)
	end
	for key in pairs(repairing) do repairing[key] = nil end
	for player in pairs(profiles) do setFlashlight(player, false) end
	clearMonsters()
	local anySurvivor = false
	for player, profile in pairs(profiles) do
		if profile.caught then
			profile.caught = false
			profile.failed = true
			profile.data.cash = math.max(0, profile.data.cash - profile.today.earned)
			profile.data.streak = 0
			setFrozen(player, false)
		end
		if not profile.failed then anySurvivor = true end
	end
	return anySurvivor
end

endNight = function()
	state.nightOver = true
end

local function runSummary()
	setPhase("Morning", Config.SummaryLength)
	applyLighting(false)
	setLights(true)
	for _, entry in pairs(entries) do refreshPrompt(entry) end
	local perfect = state.fakesServedToday == 0
	for player, profile in pairs(profiles) do
		local data = profile.data
		local survived = not profile.failed
		local bonus = 0
		if survived then
			data.streak = data.streak + 1
			data.bestNight = math.max(data.bestNight, data.streak)
			bonus = Config.SurviveBonusPerStreak * data.streak
			if perfect then bonus = bonus + Config.PerfectDayBonus end
			data.cash = data.cash + bonus
			if data.streak >= 10 then awardBadge(player, "SurviveNight10") end
			if data.streak >= 25 then awardBadge(player, "SurviveNight25") end
		end
		if perfect then awardBadge(player, "PerfectDay") end
		awardBadge(player, "FirstDay")
		data.tutorialDone = true
		data.day = state.day + 1
		data.highestDay = math.max(data.highestDay, data.day)
		publish(player)
		send(player, "Summary", {
			day = state.day,
			earned = math.floor(profile.today.earned),
			served = profile.today.served,
			caught = profile.today.caught,
			missed = state.fakesServedToday,
			survived = survived,
			bonus = bonus,
			perfect = perfect,
			streak = data.streak,
			lost = profile.today.lost or 0,
		})
		task.spawn(saveData, player)
	end
	sendAll("Sound", "MorningChime")
	task.wait(Config.SummaryLength)
end

local function grantProduct(player, key)
	local profile = profiles[player]
	if not profile then return false end
	local data = profile.data
	if key == "Revive" then
		if not revivePlayer(player) then
			data.cash = data.cash + 25 * math.max(1, data.highestDay)
			toast(player, "Revive wasn't needed, you got cash instead!", "good")
		end
	elseif key == "FakeDetector" then
		data.detectors = data.detectors + Config.DetectorPack
		toast(player, "+" .. Config.DetectorPack .. " Fake Detectors!", "good")
	elseif key == "SkipNight" then
		if state.phase == "Night" then
			state.survivedBySkip = true
			state.nightOver = true
		else
			data.detectors = data.detectors + 1
			toast(player, "It's not night, you got a Fake Detector instead!", "good")
		end
	else
		local info = Config.Products[key]
		if not info or not info.cash then return false end
		local amount = info.cash * math.max(1, data.highestDay)
		data.cash = data.cash + amount
		send(player, "Cash", amount, "Cash pack!")
	end
	publish(player)
	return true
end

local PRODUCT_BY_ID = {}

MarketplaceService.ProcessReceipt = function(receipt)
	local player = Players:GetPlayerByUserId(receipt.PlayerId)
	local profile = player and profiles[player]
	if not profile then return Enum.ProductPurchaseDecision.NotProcessedYet end
	for _, id in ipairs(profile.data.receipts) do
		if id == receipt.PurchaseId then return Enum.ProductPurchaseDecision.PurchaseGranted end
	end
	local key = PRODUCT_BY_ID[receipt.ProductId]
	if not key then return Enum.ProductPurchaseDecision.NotProcessedYet end
	local ok = pcall(grantProduct, player, key)
	if not ok then return Enum.ProductPurchaseDecision.NotProcessedYet end
	table.insert(profile.data.receipts, receipt.PurchaseId)
	if profile.canSave and not saveData(player) then
		return Enum.ProductPurchaseDecision.NotProcessedYet
	end
	return Enum.ProductPurchaseDecision.PurchaseGranted
end

local function promptProduct(player, key)
	local info = Config.Products[key]
	if not info then return end
	if info.id == 0 then
		if IS_STUDIO then

		end
		return

	if key == "Revive" and profiles[player] then profiles[player].reviving = true end
	MarketplaceService:PromptProductPurchase(player, info.id)
end

local function buyUpgrade(player, id)
	local profile = profiles[player]
	if type(id) ~= "string" or not profile then return end
	local data = profile.data
	local flavor = FLAVOR_BY_ID[id]
	local upgrade = UPGRADE_BY_ID[id]
	if not flavor and not upgrade then return end
	local owned = flavor and data.flavors[id] or (upgrade and data.upgrades[id])
	if owned then
		toast(player, "You already own that!", "info")
		return
	end
	local cost = flavor and flavor.cost or upgrade.cost
	if data.cash < cost then
		toast(player, "Not enough cash!", "bad")
		return
	end
	data.cash = data.cash - cost
	if flavor then data.flavors[id] = true else data.upgrades[id] = true end
	send(player, "Bought", flavor and flavor.name or upgrade.name)
	refreshShop()
	for _, entry in pairs(entries) do refreshPrompt(entry) end
end

local requestTimes = {}
RequestRemote.OnServerEvent:Connect(function(player, action, argument)
	local profile = profiles[player]
	if not profile or type(action) ~= "string" then return end
	local now = os.clock()
	if requestTimes[player] and now - requestTimes[player] < 0.25 then return end
	requestTimes[player] = now
	local data = profile.data
	if action == "Daily" then
		local ready, streak = dailyInfo(data)
		if not ready then return end
		data.lastDaily = today()
		data.dailyStreak = streak
		local amount = Config.DailyRewardBase * streak
		data.cash = data.cash + amount
		send(player, "Cash", amount, "Daily reward! Day " .. streak)
		publish(player)
	elseif action == "Franchise" then
		if data.cash < Config.FranchiseCost then return end
		data.cash = 0
		data.upgrades = {}
		data.flavors = {}
		for _, id in ipairs(Config.StartFlavors) do data.flavors[id] = true end
		data.franchises = data.franchises + 1
		awardBadge(player, "FirstFranchise")
		send(player, "Franchised", data.franchises)
		refreshShop()
	elseif action == "Revive" then
		if profile.caught then promptProduct(player, "Revive") end
	elseif action == "GiveUp" then
		if profile.caught then failPlayer(player) end
	elseif action == "SkipQuiet" then
		if state.phase == "Night" and state.skipAvailable then
			state.nightOver = true
		end
	elseif action == "BuyProduct" then
		if type(argument) == "string" and Config.Products[argument] then
			if argument == "Revive" and not profile.caught then return end
			promptProduct(player, argument)
		end
	elseif action == "BuyPass" then
		local info = type(argument) == "string" and Config.GamePasses[argument]
		if not info then return end
		if hasPass(player, argument) then
			toast(player, "You already own " .. info.name, "info")
		elseif info.id == 0 then
			toast(player, info.name .. " is not set up yet", "bad")
		else
			MarketplaceService:PromptGamePassPurchase(player, info.id)
		end
	elseif action == "Detect" then
		local customer = currentCustomer()
		if state.phase ~= "Day" or not customer or customer.detected then return end
		if data.detectors <= 0 then
			promptProduct(player, "FakeDetector")
			return
		end
		data.detectors = data.detectors - 1
		customer.detected = true
		make("Highlight", {
			FillColor = customer.fake and Color3.fromRGB(255, 40, 40) or Color3.fromRGB(60, 255, 100),
			OutlineColor = WHITE,
			FillTransparency = 0.35,
		}, customer.model)
		publish(player)
	end
end)

		return
	end

		if IS_STUDIO then
			canSave = false
		else
			player:Kick("We couldn't load your save. Please rejoin in a minute so your progress is safe!")
			return
		end

		data = sanitize(raw), canSave = canSave, battery = Config.FlashlightBattery,
		today = { earned = 0, served = 0, caught = 0, missed = 0 }, flashOn = false,

	profiles[player] = profile
	setupLeaderstats(player, profile)

		state.day = math.max(1, profile.data.day)
		firstDayDecided = true
	end

		task.delay(3, function() toast(player, "Saving is off in Studio (turn on API access to test saving)", "info") end)

		task.wait(0.3)
		if profile.failed then
			teleport(player, LOBBY_SPOT)
		else
			teleport(player, COUNTER_SPOT)
		end
	end)
	if player.Character then teleport(player, COUNTER_SPOT) end
	if state.phase == "Night" then
		profile.failed = true
		teleport(player, LOBBY_SPOT)
		toast(player, "A night is happening! You'll join in the morning.", "info")
	end
end

local nameCache = {}
local function refreshBoard()
	if not bestStore then return end
	local ok, pages = pcall(function() return bestStore:GetSortedAsync(false, Config.LeaderboardSize) end)
	if not ok or not pages then return end
	local entriesList = pages:GetCurrentPage()
	for _, child in ipairs(boardList:GetChildren()) do
		if child:IsA("TextLabel") then child:Destroy() end
	end
	for rank, item in ipairs(entriesList) do
		local userId = tonumber(item.key)
		local name = nameCache[userId]
		if not name and userId then
			local okName, result = pcall(function() return Players:GetNameFromUserIdAsync(userId) end)
			name = okName and result or ("Player " .. item.key)
			nameCache[userId] = name
		end
		local row = make("TextLabel", {
			Size = UDim2.new(1, -10, 0, 30),
			BackgroundColor3 = rank % 2 == 0 and Color3.fromRGB(70, 55, 90) or Color3.fromRGB(85, 65, 110),
			Text = "#" .. rank .. "   " .. (name or "?") .. "   " .. item.value,
			TextScaled = true,
			Font = Enum.Font.FredokaOne,
			TextColor3 = rank <= 3 and Color3.fromRGB(255, 215, 90) or WHITE,
			LayoutOrder = rank,
		}, boardList)
		make("UICorner", { CornerRadius = UDim.new(0, 8) }, row)
	end
end

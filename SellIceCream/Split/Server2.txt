local function refreshEntryAttributes(entry)
	gameState:SetAttribute("Entry_" .. entry.name .. "_State", entry.state)
	gameState:SetAttribute("Entry_" .. entry.name .. "_Health", math.floor(entry.health))
	gameState:SetAttribute("Entry_" .. entry.name .. "_Max", entry.maxHealth)
end

local function setEntryVisual(entry)
	if entry.kind == "door" then
		local closedCF = entry.hinge * CFrame.new(entry.width / 2, 4.5, 0)
		local target
		if entry.state == "Open" then
			target = entry.hinge * CFrame.Angles(0, math.rad(entry.name == "Back" and 100 or -100), 0) * CFrame.new(entry.width / 2, 4.5, 0)
			entry.door.CanCollide = false
			entry.door.Transparency = 0
		elseif entry.state == "Broken" then
			target = entry.hinge * CFrame.new(entry.width / 2, 0.4, 3) * CFrame.Angles(math.rad(-88), 0, 0)
			entry.door.CanCollide = false
			entry.door.Transparency = 0.2
		else
			target = closedCF
			entry.door.CanCollide = true
			entry.door.Transparency = 0
		end
		TweenService:Create(entry.door, TweenInfo.new(0.35, Enum.EasingStyle.Back), { CFrame = target }):Play()
		local offset = target * CFrame.new(entry.width / 2 - 0.8, 0, 0.5)
		entry.knob.CFrame = offset
		entry.lockLight.CFrame = target * CFrame.new(entry.width / 2 - 0.8, 1.2, 0.45)
		entry.lockLight.Color = entry.state == "Locked" and Color3.fromRGB(90, 230, 110) or RED
		entry.lockLight.Transparency = entry.state == "Broken" and 1 or 0
		entry.knob.Transparency = entry.state == "Broken" and 1 or 0
	else
		entry.glass.Transparency = entry.state == "Broken" and 1 or 0.35
		entry.glass.CanCollide = entry.state ~= "Broken"
		for _, board in ipairs(entry.boards) do
			board.Transparency = entry.state == "Boarded" and 0 or 1
			board.CanCollide = entry.state == "Boarded"
		end
	end
	refreshEntryAttributes(entry)
end

local refreshPrompt

local function resetEntries()
	for _, entry in pairs(entries) do
		entry.state = entry.startState
		if entry.kind == "door" then
			entry.maxHealth = (entry.name == "Front" and shopHasUpgrade("ReinforcedDoor")) and Config.ReinforcedDoorHealth or Config.DoorHealth
		else
			entry.maxHealth = Config.WindowBoardHealth
		end
		entry.health = entry.maxHealth
		entry.crawl = 0
		setEntryVisual(entry)
		refreshPrompt(entry)
	end
end

local repairing = {}

refreshPrompt = function(entry)
	local prompt = entry.prompt
	if not prompt then return end
	prompt.Enabled = state.phase == "Night"
	local damaged = entry.health < entry.maxHealth
	if entry.kind == "door" then
		if entry.state == "Open" or entry.state == "Closed" then
			prompt.ActionText = (entry.name == "Back" and not shopHasUpgrade("BackDoorLock")) and "Needs Back Door Lock" or "Lock Door"
			prompt.HoldDuration = Config.LockHoldTime
			entry.mode = "lock"
		elseif entry.state == "Locked" and damaged then
			prompt.ActionText = "Repair"
			prompt.HoldDuration = 30
			entry.mode = "repair"
		else
			prompt.Enabled = false
			entry.mode = nil
		end
	else
		if entry.state == "Glass" then
			prompt.ActionText = shopHasUpgrade("WindowBoards") and "Board Window" or "Needs Window Boards"
			prompt.HoldDuration = Config.BoardHoldTime
			entry.mode = "board"
		elseif entry.state == "Boarded" and damaged then
			prompt.ActionText = "Repair"
			prompt.HoldDuration = 30
			entry.mode = "repair"
		else
			prompt.Enabled = false
			entry.mode = nil
		end
	end
end

for _, name in ipairs(ENTRY_ORDER) do
	local entry = entries[name]
	local anchor = part(shopFolder, Vector3.new(1, 1, 1), CFrame.new(entry.inside + Vector3.new(0, 3, 0)), WHITE, nil, { Transparency = 1, CanCollide = false, CanQuery = false })
	local prompt = make("ProximityPrompt", {
		ObjectText = entry.display,
		ActionText = "Lock",
		MaxActivationDistance = 11,
		RequiresLineOfSight = false,
		Enabled = false,
		HoldDuration = 1.5,
		Style = Enum.ProximityPromptStyle.Default,
	}, anchor)
	entry.prompt = prompt
	prompt.Triggered:Connect(function(player)
		if state.phase ~= "Night" or not profiles[player] then return end
		if entry.mode == "lock" then
			if entry.name == "Back" and not shopHasUpgrade("BackDoorLock") then
				toast(player, "Buy the Back Door Lock upgrade to lock this door!", "bad")
				return
			end
			entry.state = "Locked"
			setEntryVisual(entry)
			refreshPrompt(entry)
			toast(player, entry.display .. " LOCKED", "good")
		elseif entry.mode == "board" then
			if not shopHasUpgrade("WindowBoards") then
				toast(player, "Buy Window Boards to board up windows!", "bad")
				return
			end
			entry.state = "Boarded"
			entry.health = entry.maxHealth
			setEntryVisual(entry)
			refreshPrompt(entry)
			toast(player, entry.display .. " BOARDED", "good")
		end
	end)
	prompt.PromptButtonHoldBegan:Connect(function(player)
		if entry.mode == "repair" and state.phase == "Night" then repairing[player] = entry end
	end)
	prompt.PromptButtonHoldEnded:Connect(function(player)
		if repairing[player] == entry then repairing[player] = nil end
	end)
	setEntryVisual(entry)
end

switchPrompt.Triggered:Connect(function(player)
	if state.phase == "Night" then
		toast(player, "The power is out!", "bad")
		return
	end
	setLights(not state.lightsOn)
end)

local function randomOf(list)
	return list[rng:NextInteger(1, #list)]
end

local function setFace(model, mode)
	local head = model:FindFirstChild("Head")
	local gui = head and head:FindFirstChild("FaceGui")
	if not gui then return end
	local eyeL, eyeR, mouth = gui:FindFirstChild("EyeL"), gui:FindFirstChild("EyeR"), gui:FindFirstChild("Mouth")
	if not eyeL or not eyeR or not mouth then return end
	if mode == "happy" then
		TweenService:Create(mouth, TweenInfo.new(0.3), { Size = UDim2.fromScale(0.4, 0.1), Position = UDim2.fromScale(0.3, 0.64) }):Play()
	elseif mode == "sad" then
		TweenService:Create(mouth, TweenInfo.new(0.3), { Size = UDim2.fromScale(0.26, 0.05), Position = UDim2.fromScale(0.37, 0.72) }):Play()
	elseif mode == "angry" then
		mouth.BackgroundColor3 = Color3.fromRGB(150, 20, 30)
		TweenService:Create(mouth, TweenInfo.new(0.2), { Size = UDim2.fromScale(0.3, 0.08), Position = UDim2.fromScale(0.35, 0.68) }):Play()
	elseif mode == "creepy" then
		TweenService:Create(mouth, TweenInfo.new(2.5, Enum.EasingStyle.Sine), { Size = UDim2.fromScale(0.78, 0.09), Position = UDim2.fromScale(0.11, 0.64) }):Play()
	elseif mode == "black" then
		eyeL.BackgroundColor3 = Color3.new(0, 0, 0)
		eyeR.BackgroundColor3 = Color3.new(0, 0, 0)
		local pl, pr = eyeL:FindFirstChild("Pupil"), eyeR:FindFirstChild("Pupil")
		if pl then pl.Visible = false end
		if pr then pr.Visible = false end
	end
end

local function buildFace(model, monster)
	local head = model:FindFirstChild("Head")
	if not head then return end
	for _, child in ipairs(head:GetChildren()) do
		if child:IsA("Decal") then child:Destroy() end
	end
	local gui = make("SurfaceGui", { Name = "FaceGui", Face = Enum.NormalId.Front, CanvasSize = Vector2.new(100, 100), LightInfluence = monster and 0 or 1 }, head)
	for _, info in ipairs({ { "EyeL", 0.2 }, { "EyeR", 0.56 } }) do
		local eye = make("Frame", {
			Name = info[1],
			Size = UDim2.fromScale(0.24, 0.24),
			Position = UDim2.fromScale(info[2], 0.3),
			BackgroundColor3 = monster and Color3.fromRGB(255, 30, 40) or WHITE,
			BorderSizePixel = 0,
		}, gui)
		make("UICorner", { CornerRadius = UDim.new(1, 0) }, eye)
		if not monster then
			local pupil = make("Frame", {
				Name = "Pupil",
				Size = UDim2.fromScale(0.55, 0.55),
				Position = UDim2.fromScale(0.225, 0.225),
				BackgroundColor3 = Color3.fromRGB(30, 25, 35),
				BorderSizePixel = 0,
			}, eye)
			make("UICorner", { CornerRadius = UDim.new(1, 0) }, pupil)
		end
	end
	local mouth = make("Frame", {
		Name = "Mouth",
		Size = monster and UDim2.fromScale(0.6, 0.14) or UDim2.fromScale(0.32, 0.06),
		Position = monster and UDim2.fromScale(0.2, 0.66) or UDim2.fromScale(0.34, 0.68),
		BackgroundColor3 = Color3.fromRGB(60, 25, 35),
		BorderSizePixel = 0,
	}, gui)
	make("UICorner", { CornerRadius = UDim.new(1, 0) }, mouth)
end

local function scaleLimbs(model, names, factor)
	local set = {}
	for _, name in ipairs(names) do set[name] = true end
	for _, joint in ipairs(model:GetDescendants()) do
		if joint:IsA("Motor6D") then
			if joint.Part0 and set[joint.Part0.Name] then
				local p = joint.C0.Position
				joint.C0 = CFrame.new(p.X, p.Y * factor, p.Z) * (joint.C0 - joint.C0.Position)
			end
			if joint.Part1 and set[joint.Part1.Name] then
				local p = joint.C1.Position
				joint.C1 = CFrame.new(p.X, p.Y * factor, p.Z) * (joint.C1 - joint.C1.Position)
			end
		end
	end
	for _, name in ipairs(names) do
		local limb = model:FindFirstChild(name)
		if limb and limb:IsA("BasePart") then
			limb.Size = Vector3.new(limb.Size.X, limb.Size.Y * factor, limb.Size.Z)
		end
	end
end

local function buildRig(kind)
	local desc = Instance.new("HumanoidDescription")
	local skin, shirt, pants
	if kind == "monster" then
		skin = Color3.fromRGB(70, 72, 80)
		shirt = Color3.fromRGB(45, 45, 55)
		pants = Color3.fromRGB(35, 35, 42)
		desc.HeightScale = Config.MonsterHeightScale
		desc.WidthScale = 0.85
		desc.DepthScale = 0.85
	else
		skin = randomOf(SKINS)
		shirt = randomOf(CLOTHES)
		pants = randomOf(CLOTHES)
	end
	desc.HeadColor = skin
	desc.LeftArmColor = skin
	desc.RightArmColor = skin
	desc.TorsoColor = shirt
	desc.LeftLegColor = pants
	desc.RightLegColor = pants
	local ok, model = pcall(function()
		return Players:CreateHumanoidModelFromDescription(desc, Enum.HumanoidRigType.R15)
	end)
	if not ok or not model then return nil end
	local animate = model:FindFirstChild("Animate")
	if animate then animate:Destroy() end
	local hum = model:FindFirstChildOfClass("Humanoid")
	local animator = hum:FindFirstChildOfClass("Animator")
	if animator then animator:Destroy() end
	hum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None
	hum.BreakJointsOnDeath = false
	hum:SetStateEnabled(Enum.HumanoidStateType.Dead, false)
	hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
	hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
	model:SetAttribute("Skin", skin)
	buildFace(model, kind == "monster")
	if kind ~= "monster" and rng:NextNumber() < 0.85 then
		local head = model:FindFirstChild("Head")
		if head then
			local hair = part(model, Vector3.new(head.Size.X * 1.08, head.Size.Y * 0.35, head.Size.Z * 1.08), head.CFrame * CFrame.new(0, head.Size.Y * 0.42, 0.05), randomOf(HAIR), nil, { Anchored = false, CanCollide = false, Massless = true })
			make("WeldConstraint", { Part0 = head, Part1 = hair }, hair)
			hair.Name = "Hair"
		end
	end
	if kind == "monster" then
		scaleLimbs(model, ARM_PARTS, Config.MonsterArmScale)
		local head = model:FindFirstChild("Head")
		if head then make("PointLight", { Color = Color3.fromRGB(255, 30, 30), Range = 9, Brightness = 2 }, head) end
	end
	for _, p in ipairs(model:GetDescendants()) do
		if p:IsA("BasePart") then p.CollisionGroup = "NPC" end
	end
	return model
end

local function placeRig(model, folder, cf)
	model:PivotTo(cf)
	model.Parent = folder
	local root = model:FindFirstChild("HumanoidRootPart")
	if root then pcall(function() root:SetNetworkOwner(nil) end) end
end

local function walkTo(model, target, timeout)
	local hum = model:FindFirstChildOfClass("Humanoid")
	local root = model:FindFirstChild("HumanoidRootPart")
	if not hum or not root then return false end
	local started = os.clock()
	local lastIssue = os.clock()
	hum:MoveTo(target)
	while model.Parent do
		local flat = Vector3.new(target.X, root.Position.Y, target.Z)
		if (root.Position - flat).Magnitude < 1.6 then return true end
		if os.clock() - started > (timeout or 20) then return false end
		if os.clock() - lastIssue > 2 then
			hum:MoveTo(target)
			lastIssue = os.clock()
		end
		task.wait(0.1)
	end
	return false
end

local function fakeChance(day)
	return math.min(Config.FakeChanceMax, Config.FakeChanceStart + (day - 1) * Config.FakeChanceStep)
end

local function pickTell(day)
	local pool = {}
	if day < Config.HardTellDay then
		for _, t in ipairs(Config.EasyTells) do table.insert(pool, t) end
	end
	if day >= Config.MediumTellDay then
		for _, t in ipairs(Config.MediumTells) do table.insert(pool, t) end
	end
	if day >= Config.HardTellDay then
		for _, t in ipairs(Config.HardTells) do table.insert(pool, t) end
	end
	return randomOf(pool)
end

local function makeOrder(day, tell)
	local flavors = shopFlavors()
	local items = {}
	local first = randomOf(flavors)
	if day >= Config.TwoFlavorDay and #flavors >= 2 and rng:NextNumber() < 0.5 then
		local second = randomOf(flavors)
		while second == first do second = randomOf(flavors) end
		table.insert(items, { flavor = first, scoops = rng:NextInteger(1, 2) })
		table.insert(items, { flavor = second, scoops = rng:NextInteger(1, 2) })
	else
		table.insert(items, { flavor = first, scoops = rng:NextInteger(1, Config.MaxScoopsPerFlavor) })
	end
	local parts = {}
	for _, item in ipairs(items) do
		local name = FLAVOR_BY_ID[item.flavor].name
		if tell == "Misspell" then name = Config.Misspellings[item.flavor] or name end
		if tell == "FakeFlavor" then name = randomOf(Config.FakeFlavorNames) end
		if #items == 1 then
			table.insert(parts, item.scoops .. (item.scoops == 1 and " scoop " or " scoops ") .. name)
		else
			table.insert(parts, item.scoops .. " " .. name)
		end
	end
	return items, table.concat(parts, " + ")
end

local function showBubble(customer)
	if customer.bubble or not customer.model.Parent then return end
	local head = customer.model:FindFirstChild("Head")
	if not head then return end
	local gui = make("BillboardGui", {
		Name = "OrderBubble",
		Size = UDim2.fromOffset(230, 96),
		StudsOffset = Vector3.new(0, 3.2, 0),
		AlwaysOnTop = true,
		MaxDistance = 70,
		LightInfluence = 0,
	}, head)
	local bubble = make("Frame", { Size = UDim2.new(1, 0, 0, 62), BackgroundColor3 = WHITE, BorderSizePixel = 0 }, gui)
	make("UICorner", { CornerRadius = UDim.new(0, 16) }, bubble)
	make("UIStroke", { Color = DARK, Thickness = 3 }, bubble)
	local label = make("TextLabel", {
		Size = UDim2.new(1, -14, 1, -8),
		Position = UDim2.fromOffset(7, 4),
		BackgroundTransparency = 1,
		Text = customer.display,
		TextScaled = true,
		TextWrapped = true,
		Font = Enum.Font.FredokaOne,
		TextColor3 = DARK,
	}, bubble)
	local back = make("Frame", { Size = UDim2.new(0.8, 0, 0, 14), Position = UDim2.new(0.1, 0, 0, 72), BackgroundColor3 = DARK, BorderSizePixel = 0 }, gui)

	make("UICorner", { CornerRadius = UDim.new(1, 0) }, bar)
	customer.bubble = gui
	customer.bubbleLabel = label
	customer.patienceBar = bar
	customer.patienceBack = back
	back.Visible = false
end

local function currentCustomer()
	local first = state.line[1]
	if first and first.atCounter and not first.leaving then return first end
	return nil
end

local function updateOrderScreen()
	local c = currentCustomer()
	if state.phase ~= "Day" then
		orderScreenText.Text = state.phase == "Night" and "CLOSED" or "..."
	elseif c then
		orderScreenText.Text = c.display
	else
		orderScreenText.Text = "NEXT!"
	end
	gameState:SetAttribute("CurrentOrder", c and c.display or "")
	gameState:SetAttribute("CurrentCustomerId", c and c.id or 0)
end

local customerId = 0

local function removeFromLine(customer)
	for i, other in ipairs(state.line) do
		if other == customer then
			table.remove(state.line, i)
			break
		end
	end
	updateOrderScreen()
end

local function customerLeave(customer, speed, mood)
	if customer.leaving then return end
	customer.leaving = true
	customer.atCounter = false
	removeFromLine(customer)
	if customer.patienceBack then customer.patienceBack.Visible = false end
	if mood then setFace(customer.model, mood) end
	task.spawn(function()
		local hum = customer.model:FindFirstChildOfClass("Humanoid")
		if hum then hum.WalkSpeed = speed or 12 end
		task.wait(0.6)
		if customer.bubble then customer.bubble:Destroy() end
		walkTo(customer.model, SP(0, 1, -19), 8)
		walkTo(customer.model, STREET_FRONT, 8)
		walkTo(customer.model, STREET_SPAWN, 25)
		customer.model:Destroy()
		state.customers[customer.id] = nil
	end)
end

local tutorialPlayers = {}

local function tutorialStep(step)
	for player in pairs(tutorialPlayers) do
		if player.Parent then send(player, "Tutorial", step) end
	end
end

local function spawnCustomer()
	state.spawnedToday = state.spawnedToday + 1
	local forcedFake = state.tutorial and state.spawnedToday == Config.TutorialFakeNumber
	local fake = forcedFake or (not state.tutorial or state.spawnedToday > Config.TutorialFakeNumber) and rng:NextNumber() < fakeChance(state.day)
	if state.tutorial and state.spawnedToday < Config.TutorialFakeNumber then fake = false end
	local tell = nil
	if fake then tell = forcedFake and "BlackEyes" or pickTell(state.day) end
	local model = buildRig("customer")
	if not model then return end
	customerId = customerId + 1
	local items, display = makeOrder(state.day, tell)
	local customer = {
		id = customerId, model = model, fake = fake, tell = tell, items = items, display = display,
		atCounter = false, leaving = false, patience = Config.Patience, tutorialFake = forcedFake,
	}
	model.Name = "Customer"
	model:SetAttribute("CustomerId", customer.id)
	local hum = model:FindFirstChildOfClass("Humanoid")
	hum.WalkSpeed = 12
	if tell == "BlackEyes" then setFace(model, "black") end
	if tell == "Float" then hum.HipHeight = hum.HipHeight + Config.FloatHeight end
	if tell == "Still" then model:SetAttribute("Still", true) end
	if tell == "HeadFollow" then model:SetAttribute("HeadFollow", true) end
	if tell == "NoShadow" then
		for _, p in ipairs(model:GetDescendants()) do
			if p:IsA("BasePart") then p.CastShadow = false end
		end
	end
	if tell == "LongArms" then scaleLimbs(model, ARM_PARTS, Config.LongArmScale) end
	placeRig(model, customerFolder, CFrame.new(STREET_SPAWN + Vector3.new(0, 3, 0)))
	state.customers[customer.id] = customer
	table.insert(state.line, customer)
	if tell == "EarlyBubble" then showBubble(customer) end
	if tell == "Flicker" then
		task.spawn(function()
			local skin = model:GetAttribute("Skin")
			while model.Parent and not customer.leaving do
				task.wait(rng:NextNumber(4, 6))
				if not model.Parent then break end
				local limbs = {}
				for _, name in ipairs({ "Head", "LeftUpperArm", "LeftLowerArm", "LeftHand", "RightUpperArm", "RightLowerArm", "RightHand" }) do
					local limb = model:FindFirstChild(name)
					if limb then
						limb.Color = Color3.fromRGB(130, 130, 135)
						table.insert(limbs, limb)
					end
				end
				task.wait(0.2)
				for _, limb in ipairs(limbs) do
					if limb.Parent then limb.Color = skin end
				end
			end
		end)
	end
	task.spawn(function()
		walkTo(model, STREET_FRONT, 30)
		while model.Parent and not customer.leaving do
			local index = 0
			for i, other in ipairs(state.line) do
				if other == customer then index = i break end
			end
			if index == 0 then break end
			local target = LINE_SLOTS[math.min(index, #LINE_SLOTS)]
			local root = model:FindFirstChild("HumanoidRootPart")
			if root and (Vector3.new(target.X, root.Position.Y, target.Z) - root.Position).Magnitude > 1.6 then
				walkTo(model, target, 3)
			elseif index == 1 and not customer.atCounter then
				customer.atCounter = true
				customer.arrivedAt = os.clock()
				if root then model:PivotTo(CFrame.lookAt(root.Position, root.Position + Vector3.new(0, 0, 1))) end
				showBubble(customer)
				if customer.bubbleLabel then customer.bubbleLabel.Text = customer.display end
				if customer.patienceBack then customer.patienceBack.Visible = true end
				updateOrderScreen()
				if customer.tutorialFake then tutorialStep("refuse") end
				if customer.tell == "SaysName" then
					task.delay(1.5, function()
						local players = Players:GetPlayers()
						if #players > 0 and customer.bubbleLabel and not customer.leaving then
							customer.bubbleLabel.Text = customer.display .. "\n..." .. randomOf(players).Name .. "..."
						end
					end)
				end
			end
			task.wait(0.15)
		end
	end)
end

local serveCooldown = {}

local function finishCustomer(player, customer, served)
	local profile = profiles[player]
	if customer.fake then
		if served then
			state.fakesServedToday = state.fakesServedToday + 1
			if profile then profile.today.missed = profile.today.missed + 1 end
			setFace(customer.model, "creepy")
			sendAll("Sound", "FakeServed")
			customerLeave(customer, 6, nil)
		end
	end
end

local function serveCustomer(player, customer, orderMap, auto)
	local profile = profiles[player]
	if not profile then return end
	if customer.fake then
		finishCustomer(player, customer, true)
		if profile then profile.today.served = profile.today.served + 1 end
		return
	end
	local correct = true
	local expected = {}
	for _, item in ipairs(customer.items) do expected[item.flavor] = (expected[item.flavor] or 0) + item.scoops end
	if not auto then
		for flavor, count in pairs(expected) do
			if orderMap[flavor] ~= count then correct = false end
		end
		for flavor, count in pairs(orderMap) do
			if expected[flavor] ~= count then correct = false end
		end
	end
	if not correct then
		toast(player, "Wrong order!", "bad")
		customerLeave(customer, 14, "angry")
		return
	end
	local base = 0
	for flavor, count in pairs(expected) do base = base + FLAVOR_BY_ID[flavor].price * count end
	local tip = 0
	if os.clock() - (customer.arrivedAt or os.clock()) <= Config.TipTime then
		local rate = Config.TipBonus
		if shopHasUpgrade("TipJar") then rate = rate + Config.TipJarBonus end
		tip = base * rate
	end
	local total = math.floor((base + tip) * cashMultiplier(player) + 0.5)
	addCash(player, total, tip > 0 and "Tip!" or nil)
	profile.today.served = profile.today.served + 1
	customerLeave(customer, 12, "happy")
end

ServeRemote.OnServerEvent:Connect(function(player, orderMap)
	local profile = profiles[player]
	if not profile or state.phase ~= "Day" or type(orderMap) ~= "table" then return end
	local now = os.clock()
	local cooldown = shopHasUpgrade("FastScoop") and Config.FastServeTime or Config.ServeTime
	if serveCooldown[player] and now - serveCooldown[player] < cooldown * 0.8 then return end
	serveCooldown[player] = now
	local clean = {}
	local total = 0
	local owned = {}
	for _, id in ipairs(shopFlavors()) do owned[id] = true end
	for flavor, count in pairs(orderMap) do
		if type(flavor) ~= "string" or not owned[flavor] or type(count) ~= "number" or count ~= math.floor(count) or count < 1 or count > Config.MaxScoopsPerFlavor then
			return
		end
		clean[flavor] = count
		total = total + count
	end
	if total < 1 or total > Config.MaxScoopsTotal then return end
	local customer = currentCustomer()
	if not customer then
		toast(player, "No customer at the counter!", "bad")
		return
	end
	local character = player.Character
	local root = character and character:FindFirstChild("HumanoidRootPart")
	if not root or (root.Position - SP(0, 3, 0)).Magnitude > 40 then
		toast(player, "Get behind your counter!", "bad")
		return
	end
	serveCustomer(player, customer, clean, false)
end)

RefuseRemote.OnServerEvent:Connect(function(player)
	local profile = profiles[player]
	if not profile or state.phase ~= "Day" then return end
	local now = os.clock()
	if serveCooldown[player] and now - serveCooldown[player] < 0.4 then return end
	serveCooldown[player] = now
	local customer = currentCustomer()
	if not customer then return end
	if customer.fake then
		profile.data.fakesCaught = profile.data.fakesCaught + 1
		profile.today.caught = profile.today.caught + 1
		state.fakesCaughtToday = state.fakesCaughtToday + 1
		if profile.data.fakesCaught >= 10 then awardBadge(player, "Catch10") end
		if profile.data.fakesCaught >= 100 then awardBadge(player, "Catch100") end
		addCash(player, math.floor(Config.RefuseFakeBonus * cashMultiplier(player)), "Fake caught!")
		local root = customer.model:FindFirstChild("HumanoidRootPart")
		if root then soundIn(root, "Hiss", 1) end
		sendAll("Hiss", customer.id)
		task.spawn(function()
			local head = customer.model:FindFirstChild("Head")
			local gui = head and head:FindFirstChild("FaceGui")
			for _ = 1, 10 do
				if gui then
					for _, frame in ipairs(gui:GetChildren()) do
						if frame:IsA("Frame") then
							frame.Position = frame.Position + UDim2.fromScale(rng:NextNumber(-0.08, 0.08), rng:NextNumber(-0.08, 0.08))
							frame.BackgroundColor3 = rng:NextNumber() < 0.5 and Color3.new(0, 0, 0) or RED
						end
					end
				end
				task.wait(0.05)
			end
		end)
		customerLeave(customer, 30, nil)
	else
		profile.data.cash = math.max(0, profile.data.cash - Config.RefuseRealPenalty)
		publish(player)
		send(player, "Cash", -Config.RefuseRealPenalty, "Complaint!")
		customerLeave(customer, 10, "sad")
	end
	publish(player)
end)

local function teleport(player, cf)
	local character = player.Character
	if character and character:FindFirstChild("HumanoidRootPart") then
		character:PivotTo(cf)
	end
end

local function playersInGame()
	local list = {}
	for player in pairs(profiles) do
		if player.Parent then table.insert(list, player) end
	end
	return list
end

local function runDay()
	setPhase("Day", Config.DayLength)
	applyLighting(false)
	setLights(true)
	state.line = {}
	state.spawnedToday = 0
	state.fakesServedToday = 0
	state.fakesCaughtToday = 0
	tutorialPlayers = {}
	state.tutorial = false
	for player, profile in pairs(profiles) do
		profile.today = { earned = 0, served = 0, caught = 0, missed = 0 }
		profile.failed = false
		profile.caught = false
		if not profile.data.tutorialDone then
			tutorialPlayers[player] = true
			state.tutorial = true
		end
	end
	state.tutorialNight = state.tutorial
	for _, entry in pairs(entries) do
		entry.state = entry.startState
		if entry.name == "Front" then entry.state = "Open" end
	end
	resetEntries()
	for _, player in ipairs(playersInGame()) do
		teleport(player, COUNTER_SPOT)
		local hum = player.Character and player.Character:FindFirstChildOfClass("Humanoid")
		if hum then hum.WalkSpeed = Config.PlayerSpeed end
	end
	updateOrderScreen()
	sendAll("PhaseStart", "Day", state.day)
	if state.tutorial then
		task.delay(1.5, function() tutorialStep("flavors") end)
	end
	local nextSpawn = os.clock() + 2
	local autoTimers = {}
	while workspace:GetServerTimeNow() < state.endsAt do
		if #playersInGame() == 0 then break end
		local now = os.clock()
		local interval = math.max(Config.SpawnIntervalMin, Config.SpawnIntervalStart - (state.day - 1) * Config.SpawnIntervalStep)
		if shopHasUpgrade("NeonSign") then interval = interval / (1 + Config.NeonSignBonus) end
		if now >= nextSpawn and #state.line < Config.MaxLine then
			nextSpawn = now + interval
			task.spawn(spawnCustomer)
		end
		for _, customer in pairs(state.customers) do
			if customer.atCounter and not customer.leaving then
				local left = Config.Patience - (now - customer.arrivedAt)
				if customer.patienceBar then
					local ratio = math.clamp(left / Config.Patience, 0, 1)
					customer.patienceBar.Size = UDim2.new(ratio, -4, 1, -4)
					customer.patienceBar.BackgroundColor3 = ratio > 0.5 and Color3.fromRGB(110, 230, 110) or (ratio > 0.25 and Color3.fromRGB(255, 210, 60) or Color3.fromRGB(240, 70, 70))
				end
				if left <= 0 then
					customerLeave(customer, 14, "angry")
				else
					local autoOwner = anyoneHasPass("AutoWorker")
					if autoOwner and now - customer.arrivedAt >= Config.AutoWorkerDelay and not autoTimers[customer.id] then
						autoTimers[customer.id] = true
						serveCustomer(autoOwner, customer, nil, true)
					end
				end
			end
		end
		task.wait(0.2)
	end
	for _, customer in pairs(state.customers) do
		customerLeave(customer, 16, nil)
	end
	task.wait(1)
end

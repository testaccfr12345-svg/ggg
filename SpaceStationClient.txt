local function new(className, props, parent)
	local instance = Instance.new(className)
	for key, value in pairs(props) do
		instance[key] = value
	end
	instance.Parent = parent
	return instance
end

local function tween(object, time, props, style, direction)
	local t = TweenService:Create(object, TweenInfo.new(time, style or Enum.EasingStyle.Quad, direction or Enum.EasingDirection.Out), props)
	t:Play()
	return t
end

local function formatNumber(n)
	n = math.floor(n or 0)
	if n < 1000 then return tostring(n) end
	local suffixes = { "", "K", "M", "B", "T", "Qa", "Qi" }
	local index = 1
	local value = n
	while value >= 1000 and index < #suffixes do
		value = value / 1000
		index += 1
	end
	if value >= 100 then
		return string.format("%.0f%s", value, suffixes[index])
	elseif value >= 10 then
		return string.format("%.1f%s", value, suffixes[index])
	end
	return string.format("%.2f%s", value, suffixes[index])
end

local function label(parent, text, size, position, textSize, color, font, align)
	return new("TextLabel", {

end

local function pressable(button)
	local scale = new("UIScale", { Scale = 1 }, button)
	button.MouseButton1Down:Connect(function() tween(scale, 0.08, { Scale = 0.92 }) end)
	button.MouseButton1Up:Connect(function() tween(scale, 0.25, { Scale = 1 }, Enum.EasingStyle.Back) end)
	button.MouseLeave:Connect(function() tween(scale, 0.2, { Scale = 1 }) end)
	return scale
end

local toastCount = 0

local TOAST_COLORS = { good = GREEN, bad = RED, info = CYAN, build = YELLOW, rebirth = PURPLE }

local function toast(text, kind)
	toastCount += 1
	local color = TOAST_COLORS[kind] or CYAN
	local frame = new("Frame", {
		Size = UDim2.fromOffset(380, 34),
		BackgroundColor3 = PANEL,
		BackgroundTransparency = 0.12,
		BorderSizePixel = 0,
		LayoutOrder = toastCount,
		ZIndex = 6,
	}, toasts)
	new("UICorner", { CornerRadius = UDim.new(0, 6) }, frame)
	new("UIStroke", { Color = color, Thickness = 1.5, Transparency = 0.2 }, frame)
	local textLabel = label(frame, text, UDim2.new(1, -20, 1, 0), UDim2.fromOffset(10, 0), 13, color, BODY)
	textLabel.ZIndex = 7
	textLabel.TextWrapped = true
	local scale = new("UIScale", { Scale = 0.6 }, frame)
	tween(scale, 0.3, { Scale = 1 }, Enum.EasingStyle.Back)
	task.delay(2.6, function()
		if not frame.Parent then return end
		tween(frame, 0.3, { BackgroundTransparency = 1 })
		tween(textLabel, 0.3, { TextTransparency = 1 })
		tween(scale, 0.3, { Scale = 0.8 })
		task.wait(0.3)
		frame:Destroy()
	end)
	local children = {}
	for _, child in ipairs(toasts:GetChildren()) do
		if child:IsA("Frame") then table.insert(children, child) end
	end
	if #children > 4 then children[1]:Destroy() end
end

local function floatText(text, color, fromObject)
	local position = fromObject.AbsolutePosition + fromObject.AbsoluteSize / 2 - root.AbsolutePosition
	local s = uiScale.Scale
	local floater = label(root, text, UDim2.fromOffset(160, 30), UDim2.fromOffset(position.X / s - 80, position.Y / s - 15), 22, color, HEAD)
	floater.ZIndex = 9
	floater.TextStrokeTransparency = 0.2
	local drift = math.random(-40, 40)
	tween(floater, 0.8, {
		Position = floater.Position + UDim2.fromOffset(drift, -90),
		TextTransparency = 1,
		TextStrokeTransparency = 1,
	})
	task.delay(0.85, function() floater:Destroy() end)
end

local openPanel = nil

local function closePanel()
	if not openPanel then return end
	local panel = openPanel
	openPanel = nil
	local scale = panel:FindFirstChildOfClass("UIScale")
	tween(overlay, 0.2, { BackgroundTransparency = 1 })
	tween(scale, 0.18, { Scale = 0.85 }).Completed:Connect(function()
		if openPanel ~= panel then
			panel.Visible = false
			if not openPanel then overlay.Visible = false end
		end
	end)
end

local function showPanel(panel)
	if openPanel == panel then closePanel() return end
	if openPanel then openPanel.Visible = false end
	openPanel = panel
	overlay.Visible = true
	panel.Visible = true
	local scale = panel:FindFirstChildOfClass("UIScale")
	scale.Scale = 0.7
	tween(scale, 0.3, { Scale = 1 }, Enum.EasingStyle.Back)
	tween(overlay, 0.2, { BackgroundTransparency = 0.45 })
end

overlay.Activated:Connect(closePanel)

local shownCredits = 0
local targetCredits = 0
local lastCredits = nil

local function refresh()
	local credits = player:GetAttribute("Credits") or 0
	local energy = player:GetAttribute("Energy") or 0
	local capacity = math.max(1, player:GetAttribute("Capacity") or 100)
	local rate = player:GetAttribute("Rate") or 0
	local sell = player:GetAttribute("SellValue") or 1
	local rebirths = player:GetAttribute("Rebirths") or 0
	local multiplier = player:GetAttribute("Multiplier") or 1
	local canRebirth = player:GetAttribute("CanRebirth") == true

	if lastCredits and credits > lastCredits then
		walletScale.Scale = 1.08
		tween(walletScale, 0.3, { Scale = 1 }, Enum.EasingStyle.Back)
	end
	lastCredits = credits
	targetCredits = credits

	local ratio = math.clamp(energy / capacity, 0, 1)
	tween(fill, 0.25, { Size = UDim2.fromScale(ratio, 1) })
	fillGradient.Color = ratio >= 1 and ColorSequence.new(Color3.fromRGB(200, 40, 60), RED) or ColorSequence.new(TEAL, CYAN)
	energyText.Text = "ENERGY " .. formatNumber(energy) .. " / " .. formatNumber(capacity)
	rateText.Text = "+" .. formatNumber(rate) .. " ENERGY / SEC"
	chargeAmount.Text = "+" .. formatNumber(player:GetAttribute("ChargePower") or 1)
	sellValue.Text = "$" .. formatNumber(energy * sell)
	autoTag.Visible = player:GetAttribute("AutoSell") == true

	for kind, row in pairs(upgradeRows) do
		local level = player:GetAttribute(kind .. "Level") or 0
		local cost = player:GetAttribute(kind .. "Cost") or 0
		row.level.Text = "LV " .. level
		row.price.Text = "$" .. formatNumber(cost)
		row.button.BackgroundColor3 = credits >= cost and GREEN or Color3.fromRGB(90, 100, 120)
	end

	rebirthStats.Text = "REBIRTHS " .. rebirths .. "     MULTIPLIER x" .. string.format("%.1f", multiplier) .. "  >  x" .. string.format("%.1f", multiplier + 0.5)
	rebirthRequirement.Text = canRebirth and "WARP GATE ONLINE - READY" or "REQUIRES: WARP GATE"
	rebirthRequirement.TextColor3 = canRebirth and GREEN or RED
	rebirthConfirm.BackgroundColor3 = canRebirth and PURPLE or Color3.fromRGB(70, 70, 90)
	rebirthDot.Visible = canRebirth
end

player.AttributeChanged:Connect(refresh)
refresh()

	creditsText.Text = "$" .. formatNumber(shownCredits)

end)

remote.OnClientEvent:Connect(function(kind, a, b)
	if kind == "Notify" then
		toast(a, b)
	elseif kind == "Built" then
		toast("BUILT: " .. string.upper(a), "build")
	elseif kind == "Sold" then
		floatText("+$" .. formatNumber(a), GREEN, b and wallet or sellButton)
		if not b then toast("Sold energy for $" .. formatNumber(a), "good") end
	elseif kind == "Upgraded" then
		toast(a .. " upgraded to LV " .. b, "good")
	elseif kind == "Rebirthed" then
		toast("REBIRTH " .. a .. "! Multiplier is now x" .. string.format("%.1f", b), "rebirth")
	end
end)

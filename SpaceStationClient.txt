local function play(name, speed)
	local sound = sounds[name]
	if not sound then return end
	sound.PlaybackSpeed = speed or 1
	sound.TimePosition = 0
	sound:Play()
end

local function new(className, props, parent)
	local instance = Instance.new(className)
	for key, value in pairs(props) do
		instance[key] = value
	end
	instance.Parent = parent
	return instance
end

local function tween(object, time, props, style, direction)
	local t = TweenService:Create(object, TweenInfo.new(time, style or Enum.EasingStyle.Quad, direction or Enum.EasingDirection.Out), props)
	t:Play()
	return t
end

local function corner(parent, radius)
	return new("UICorner", { CornerRadius = radius == "round" and UDim.new(1, 0) or UDim.new(0, radius) }, parent)
end

local function formatNumber(n)
	n = math.floor(n or 0)
	if n < 1000 then return tostring(n) end
	local suffixes = { "", "K", "M", "B", "T", "Qa", "Qi" }
	local index = 1
	local value = n
	while value >= 1000 and index < #suffixes do
		value = value / 1000
		index = index + 1
	end
	if value >= 100 then
		return string.format("%.0f%s", value, suffixes[index])
	elseif value >= 10 then
		return string.format("%.1f%s", value, suffixes[index])
	end
	return string.format("%.2f%s", value, suffixes[index])
end

local function text(parent, str, size, position, textSize, color, thickness, align)
	local label = new("TextLabel", {
		BackgroundTransparency = 1,
		Size = size,
		Position = position,
		Text = str,
		TextSize = textSize,
		TextColor3 = color or WHITE,
		Font = FONT,
		TextXAlignment = align or Enum.TextXAlignment.Center,
		ZIndex = 6,
	}, parent)
	if thickness ~= 0 then
		new("UIStroke", { Color = OUTLINE, Thickness = thickness or 2.5, LineJoinMode = Enum.LineJoinMode.Round }, label)
	end
	return label
end

local function chunky(parent, size, position, anchor, color, radius, depth)
	radius = radius or 14
	depth = depth or 6
	local holder = new("Frame", {
		Size = size,
		Position = position,
		AnchorPoint = anchor or Vector2.new(0, 0),
		BackgroundTransparency = 1,
	}, parent)
	local edge = new("Frame", {
		Size = UDim2.fromScale(1, 1),
		BackgroundColor3 = color:Lerp(BLACK, 0.42),
		BorderSizePixel = 0,
		ZIndex = 2,
	}, holder)
	corner(edge, radius)
	new("UIStroke", { Color = OUTLINE, Thickness = 3, ApplyStrokeMode = Enum.ApplyStrokeMode.Border }, edge)
	local face = new("Frame", {
		Size = UDim2.new(1, 0, 1, -depth),
		BackgroundColor3 = color,
		BorderSizePixel = 0,
		ZIndex = 3,
	}, holder)
	corner(face, radius)
	new("UIGradient", { Rotation = 90, Color = ColorSequence.new(WHITE, Color3.fromRGB(200, 200, 200)) }, face)
	local gloss = new("Frame", {
		Size = UDim2.new(1, -14, 0.36, 0),
		Position = UDim2.fromOffset(7, 5),
		BackgroundColor3 = WHITE,
		BackgroundTransparency = 0.8,
		BorderSizePixel = 0,
		ZIndex = 4,
	}, face)
	corner(gloss, radius == "round" and "round" or math.max(4, radius - 5))
	local piece = { holder = holder, edge = edge, face = face, depth = depth }
	piece.setColor = function(newColor)
		face.BackgroundColor3 = newColor
		edge.BackgroundColor3 = newColor:Lerp(BLACK, 0.42)
	end
	return piece
end

local function clickable(piece, callback)
	local button = new("TextButton", {
		Size = UDim2.fromScale(1, 1),
		BackgroundTransparency = 1,
		Text = "",
		AutoButtonColor = false,
		ZIndex = 10,
	}, piece.holder)
	local scale = new("UIScale", { Scale = 1 }, piece.holder)
	piece.scale = scale
	local function release()
		tween(piece.face, 0.12, { Position = UDim2.fromOffset(0, 0) }, Enum.EasingStyle.Back)
	end
	button.MouseButton1Down:Connect(function()
		tween(piece.face, 0.05, { Position = UDim2.fromOffset(0, piece.depth - 2) })
	end)
	button.MouseButton1Up:Connect(release)
	button.MouseLeave:Connect(function()
		release()
		tween(scale, 0.15, { Scale = 1 })
	end)
	button.MouseEnter:Connect(function()
		if UserInputService.MouseEnabled then tween(scale, 0.15, { Scale = 1.05 }, Enum.EasingStyle.Back) end
	end)
	button.Activated:Connect(function()
		callback()
	end)
	piece.button = button
	return piece
end

local function bump(piece, amount)
	if not piece.scale then return end
	piece.scale.Scale = amount or 1.12
	tween(piece.scale, 0.3, { Scale = 1 }, Enum.EasingStyle.Back)
end

end

local function toast(message, kind)
	toastCount = toastCount + 1
	local frame = new("Frame", {
		Size = UDim2.fromOffset(420, 40),
		BackgroundColor3 = TOAST_COLORS[kind] or CYAN,
		BorderSizePixel = 0,
		LayoutOrder = toastCount,
		ZIndex = 14,
	}, toasts)
	corner(frame, 14)
	new("UIStroke", { Color = OUTLINE, Thickness = 3, ApplyStrokeMode = Enum.ApplyStrokeMode.Border }, frame)
	new("UIGradient", { Rotation = 90, Color = ColorSequence.new(WHITE, Color3.fromRGB(200, 200, 200)) }, frame)
	local label = text(frame, message, UDim2.new(1, -16, 1, 0), UDim2.fromOffset(8, 0), 18, WHITE, 2.5)
	label.ZIndex = 15
	label.TextScaled = true
	new("UITextSizeConstraint", { MaxTextSize = 18, MinTextSize = 10 }, label)
	local scale = new("UIScale", { Scale = 0.4 }, frame)
	tween(scale, 0.3, { Scale = 1 }, Enum.EasingStyle.Back)
	task.delay(2.8, function()
		if not frame.Parent then return end
		tween(scale, 0.2, { Scale = 0 }, Enum.EasingStyle.Back, Enum.EasingDirection.In)
		task.wait(0.22)
		frame:Destroy()
	end)
	local list = {}
	for _, child in ipairs(toasts:GetChildren()) do
		if child:IsA("Frame") then table.insert(list, child) end
	end
	table.sort(list, function(a, b) return a.LayoutOrder < b.LayoutOrder end)
	if #list > 3 then list[1]:Destroy() end
end

end

local function doCharge()
	remote:FireServer("Charge")
	local energy = player:GetAttribute("Energy") or 0
	local capacity = math.max(1, player:GetAttribute("Capacity") or 100)
	if energy >= capacity and not player:GetAttribute("AutoSell") then
		floatText("FULL! SELL IT", RED, chargePiece.holder)
		play("Error", 0.9)
		bump(sellPiece, 1.18)
	else
		floatText("+" .. formatNumber(player:GetAttribute("ChargePower") or 1), CYAN, chargePiece.holder)
		play("Charge", 0.9 + math.clamp(energy / capacity, 0, 1) * 0.7)
	end
	bump(chargePiece, 0.9)
end

local function doSell()
	if (player:GetAttribute("Energy") or 0) < 1 then
		play("Error", 0.7)
		toast("No energy yet! Tap CHARGE first", "bad")
		return
	end
	remote:FireServer("Sell")
	bump(sellPiece, 0.9)
end

clickable(chargePiece, doCharge)
clickable(sellPiece, doSell)
clickable(upgradesPiece, function()
	play("Click")
	showPanel(upgradesPanel)
end)
clickable(rebirthPiece, function()
	play("Click")
	showPanel(rebirthPanel)
end)
clickable(homePiece, function()
	play("Whoosh", 1.2)
	remote:FireServer("Home")

UserInputService.InputBegan:Connect(function(input, processed)
	if processed then return end
	if input.KeyCode == Enum.KeyCode.E then
		doCharge()
	elseif input.KeyCode == Enum.KeyCode.Q then
		doSell()
	end
end)

local shownCredits = 0
local targetCredits = 0
local lastCredits = nil

local function refresh()
	local credits = player:GetAttribute("Credits") or 0
	local energy = player:GetAttribute("Energy") or 0
	local capacity = math.max(1, player:GetAttribute("Capacity") or 100)
	local rate = player:GetAttribute("Rate") or 0
	local sell = player:GetAttribute("SellValue") or 1
	local rebirths = player:GetAttribute("Rebirths") or 0
	local multiplier = player:GetAttribute("Multiplier") or 1
	local canRebirth = player:GetAttribute("CanRebirth") == true

	if lastCredits and credits > lastCredits then
		walletScale.Scale = 1.1
		tween(walletScale, 0.3, { Scale = 1 }, Enum.EasingStyle.Back)
	end
	lastCredits = credits
	targetCredits = credits

	local ratio = math.clamp(energy / capacity, 0, 1)
	tween(fill, 0.2, { Size = UDim2.fromScale(ratio, 1) })
	if ratio >= 1 then
		fillGradient.Color = ColorSequence.new(Color3.fromRGB(255, 150, 150), RED)
	else
		fillGradient.Color = ColorSequence.new(Color3.fromRGB(120, 240, 255), Color3.fromRGB(20, 150, 255))
	end
	energyText.Text = (ratio >= 1 and "FULL " or "ENERGY ") .. formatNumber(energy) .. " / " .. formatNumber(capacity)
	rateText.Text = "+" .. formatNumber(rate) .. " / SEC" .. (player:GetAttribute("AutoSell") and "   AUTO-SELL ON" or "")
	chargeAmount.Text = "+" .. formatNumber(player:GetAttribute("ChargePower") or 1)
	sellValue.Text = "$" .. formatNumber(energy * sell)
	sellPiece.setColor(energy >= 1 and GREEN or GREY)

	for kind, row in pairs(upgradeRows) do
		local level = player:GetAttribute(kind .. "Level") or 0
		local cost = player:GetAttribute(kind .. "Cost") or 0
		row.level.Text = tostring(level)
		row.price.Text = "$" .. formatNumber(cost)
		row.piece.setColor(credits >= cost and GREEN or GREY)
	end

	rebirthCount.Text = "REBIRTHS: " .. rebirths
	rebirthMultiplier.Text = string.format("x%.1f  >  x%.1f", multiplier, multiplier + 0.5)
	rebirthRequirement.Text = canRebirth and "WARP GATE READY!" or "NEEDS: WARP GATE"
	rebirthRequirement.TextColor3 = canRebirth and GREEN or RED
	rebirthConfirm.setColor(canRebirth and PURPLE or GREY)
	rebirthDot.Visible = canRebirth
end

player.AttributeChanged:Connect(refresh)
refresh()

RunService.RenderStepped:Connect(function(dt)
	shownCredits = shownCredits + (targetCredits - shownCredits) * math.clamp(dt * 10, 0, 1)
	if math.abs(targetCredits - shownCredits) < 1 then shownCredits = targetCredits end
	creditsText.Text = formatNumber(shownCredits)
	if rebirthDot.Visible then
		local pulse = 1 + math.sin(os.clock() * 6) * 0.12
		rebirthDot.Size = UDim2.fromOffset(22 * pulse, 22 * pulse)
	end
end)

remote.OnClientEvent:Connect(function(kind, a, b)
	if kind == "Notify" then
		toast(a, b)
		play(b == "bad" and "Error" or "Upgrade", b == "bad" and 0.7 or 1.2)
	elseif kind == "Built" then
		toast("BUILT: " .. string.upper(a) .. "!", "build")
		play("Build", 1.1)
	elseif kind == "Sold" then
		floatText("+$" .. formatNumber(a), GREEN, b and wallet.holder or sellPiece.holder)
		play("Sell", 1.1)
		if not b then coinBurst(sellPiece.holder) end
	elseif kind == "Upgraded" then
		toast("UPGRADED TO LEVEL " .. b .. "!", "good")
		play("Upgrade", 1.3)
	elseif kind == "Rebirthed" then
		toast("REBIRTH " .. a .. "! NOW x" .. string.format("%.1f", b), "rebirth")
		play("Rebirth")
		flash.BackgroundTransparency = 0
		tween(flash, 0.8, { BackgroundTransparency = 1 })
	elseif kind == "Checkpoint" then
		toast("CHECKPOINT: " .. a, "checkpoint")
		play("Charge", 1.5)
	elseif kind == "Fell" then
		toast("You fell! Back to your checkpoint", "info")
		play("Whoosh", 0.8)
	end
end)

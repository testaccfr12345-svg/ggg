local UPRIGHT = CFrame.Angles(0, 0, math.rad(90))

local function block(parent, shape, size, cf, color, material, extra)
	local p = Instance.new("Part")
	p.Anchored = true
	p.Shape = shape or Enum.PartType.Block
	p.Size = size
	p.CFrame = cf
	p.Color = color
	p.Material = material or Enum.Material.SmoothPlastic
	p.TopSurface = Enum.SurfaceType.Smooth
	p.BottomSurface = Enum.SurfaceType.Smooth
	if extra then
		for key, value in pairs(extra) do
			p[key] = value
		end
	end
	p.Parent = parent
	return p
end

local function box(parent, size, cf, color, material, extra)
	return block(parent, Enum.PartType.Block, size, cf, color, material, extra)
end

local function formatNumber(n)
	n = math.floor(n)
	if n < 1000 then return tostring(n) end
	local suffixes = { "", "K", "M", "B", "T", "Qa", "Qi" }
	local index = 1
	local value = n
	while value >= 1000 and index < #suffixes do
		value = value / 1000
		index += 1
	end
	if value >= 100 then
		return string.format("%.0f%s", value, suffixes[index])
	elseif value >= 10 then
		return string.format("%.1f%s", value, suffixes[index])
	end
	return string.format("%.2f%s", value, suffixes[index])
end

local function at(plot, x, y, z)
	return plot.cf * CFrame.new(x, 1 + y, z)
end

local function spinner(plot, part, pivot, offset, speed)
	table.insert(plot.spinners, { part = part, pivot = pivot, offset = offset, speed = speed })
end

local function orbiters(plot, model, pivot, count, radius, size, color, speed, tilt)
	for i = 1, count do
		local angle = i / count * math.pi * 2
		local offset = CFrame.Angles(0, angle, 0) * CFrame.new(radius, 0, 0)
		local p = box(model, Vector3.new(size, size, size), pivot * (tilt or CFrame.identity) * offset, color, Enum.Material.Neon, { CanCollide = false })
		spinner(plot, p, pivot * (tilt or CFrame.identity), offset, Vector3.new(0, speed, 0))
	end
end

	end
	cylinder(model, 0.6, 1.6, at(plot, rx, 0.8, rz), ORANGE, Enum.Material.Neon, { CanCollide = false })
end

	local x, z = 12, -16

	local center = at(plot, x, 6.5, z)

local ORDER = {}
for _, item in ipairs(STATION) do table.insert(ORDER, item.id) end
for _, item in ipairs(GENERATORS) do table.insert(ORDER, item.id) end

local profiles = {}

local function upgradeCost(kind, level)
	local data = UPGRADES[kind]
	return math.floor(data.base * data.growth ^ level)
end

local function stats(profile)
	local multiplier = 1 + profile.rebirths * 0.5
	local rate = 0
	for _, item in ipairs(GENERATORS) do
		if profile.items[item.id] then rate += item.rate end
	end
	local capacity = BASE_CAPACITY
	for _, item in ipairs(STATION) do
		if profile.items[item.id] and item.capacity then capacity += item.capacity end
	end
	capacity = math.floor(capacity * (1 + profile.upgrades.Capacity * 0.5))
	return {
		multiplier = multiplier,
		rate = rate * multiplier,
		capacity = capacity,
		charge = (1 + profile.upgrades.Charge * 2) * multiplier,
		sell = (1 + profile.upgrades.Sell * 0.2) * multiplier,
		autoSell = profile.items.Antenna == true,
	}
end

local function publish(player)
	local profile = profiles[player]
	if not profile then return end
	local s = stats(profile)
	profile.stats = s
	player:SetAttribute("Credits", profile.cash)
	player:SetAttribute("Energy", profile.energy)
	player:SetAttribute("Capacity", s.capacity)
	player:SetAttribute("Rate", s.rate)
	player:SetAttribute("ChargePower", s.charge)
	player:SetAttribute("SellValue", s.sell)
	player:SetAttribute("Multiplier", s.multiplier)
	player:SetAttribute("Rebirths", profile.rebirths)
	player:SetAttribute("AutoSell", s.autoSell)
	player:SetAttribute("CanRebirth", profile.items.WarpGate == true)
	for kind in pairs(UPGRADES) do
		player:SetAttribute(kind .. "Level", profile.upgrades[kind])
		player:SetAttribute(kind .. "Cost", upgradeCost(kind, profile.upgrades[kind]))
	end
	if profile.leaderCash then profile.leaderCash.Value = math.floor(profile.cash) end
	if profile.leaderRebirths then profile.leaderRebirths.Value = profile.rebirths end
end

local function notify(player, text, kind)
	remote:FireClient(player, "Notify", text, kind or "info")
end

local function isAvailable(profile, id)
	local previous = PREVIOUS[id]
	return previous == nil or profile.items[previous] == true
end

local function clearPlot(plot)
	plot.items:ClearAllChildren()
	plot.buttons:ClearAllChildren()
	plot.spinners = {}
	plot.fills = {}
	plot.blinkers = {}
	plot.buttonParts = {}
end

local function buildItem(plot, id, animate)
	local builder = BUILD[id]
	if not builder then return end
	local model = Instance.new("Model")
	model.Name = id
	model.Parent = plot.items
	builder(plot, model)
	if not animate then return end
	for _, p in ipairs(model:GetDescendants()) do
		if p:IsA("BasePart") then
			local final = p.CFrame
			local transparency = p.Transparency
			p.CFrame = final * CFrame.new(0, -5, 0)
			p.Transparency = 1
			TweenService:Create(p, TweenInfo.new(0.6, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
				CFrame = final, Transparency = transparency,
			}):Play()
		end
	end
end

local buy

local function refreshButtons(plot)
	plot.buttons:ClearAllChildren()
	plot.buttonParts = {}
	local owner = plot.owner
	local profile = owner and profiles[owner]
	if not profile then return end
	for slot, chain in ipairs({ GENERATORS, STATION }) do
		local nextItem = nil
		for _, item in ipairs(chain) do
			if not profile.items[item.id] then
				nextItem = item
				break
			end
		end
		if nextItem and isAvailable(profile, nextItem.id) then
			local x = slot == 1 and -9 or 9
			local pad = box(plot.buttons, Vector3.new(6, 0.6, 6), at(plot, x, 0.3, -33), GREEN, Enum.Material.Neon)
			local rim = box(plot.buttons, Vector3.new(7, 0.4, 7), at(plot, x, 0.2, -33), DARK, Enum.Material.Metal)
			rim.CanTouch = false
			local gui = billboard(pad, UDim2.fromOffset(220, 74), Vector3.new(0, 4, 0), 90)
			textLabel(gui, string.upper(nextItem.name), UDim2.new(1, 0, 0.5, 0), UDim2.fromScale(0, 0), 20, WHITE, Enum.Font.Michroma)
			local priceText = nextItem.price == 0 and "FREE" or ("$" .. formatNumber(nextItem.price))
			local price = textLabel(gui, priceText, UDim2.new(1, 0, 0.5, 0), UDim2.fromScale(0, 0.5), 24, GREEN, Enum.Font.GothamBlack)
			table.insert(plot.buttonParts, { pad = pad, price = price, item = nextItem })
			local itemId = nextItem.id
			pad.Touched:Connect(function(hit)
				local character = hit.Parent
				local toucher = character and Players:GetPlayerFromCharacter(character)
				if toucher and toucher == plot.owner then
					buy(toucher, itemId)
				end
			end)
		end
	end
end

buy = function(player, id)
	local profile = profiles[player]
	local item = ITEM_BY_ID[id]
	if not profile or not item or profile.items[id] or not isAvailable(profile, id) then return end
	local now = os.clock()
	if profile.lastDenied and now - profile.lastDenied < 1.5 and profile.cash < item.price then return end
	if profile.cash < item.price then
		profile.lastDenied = now
		notify(player, "Need $" .. formatNumber(item.price - profile.cash) .. " more for " .. item.name, "bad")
		return
	end
	profile.cash -= item.price
	profile.items[id] = true
	buildItem(profile.plot, id, true)
	refreshButtons(profile.plot)
	publish(player)
	remote:FireClient(player, "Built", item.name)
	if id == "WarpGate" then
		notify(player, "Warp Gate online! You can now REBIRTH", "good")
	elseif id == "Antenna" then
		notify(player, "Energy now auto-sells when your battery is full", "good")
	end
end

local function sell(player, auto)
	local profile = profiles[player]
	if not profile or profile.energy < 1 then return end
	local amount = math.floor(profile.energy * profile.stats.sell)
	profile.cash += amount
	profile.energy = 0
	publish(player)
	remote:FireClient(player, "Sold", amount, auto == true)
end

local function teleportHome(player)
	local profile = profiles[player]
	local character = player.Character
	if not profile or not profile.plot or not character then return end
	local root = character:FindFirstChild("HumanoidRootPart")
	if root then
		character:PivotTo(profile.plot.cf * CFrame.new(0, 5, -35) * CFrame.Angles(0, math.pi, 0))
	end
end

local function rebirth(player)
	local profile = profiles[player]
	if not profile or not profile.items.WarpGate then return end
	profile.rebirths += 1
	profile.cash = 0
	profile.energy = 0
	profile.items = {}
	profile.upgrades = { Capacity = 0, Charge = 0, Sell = 0 }
	clearPlot(profile.plot)
	refreshButtons(profile.plot)
	publish(player)
	remote:FireClient(player, "Rebirthed", profile.rebirths, profile.stats.multiplier)
	teleportHome(player)
end

local function upgrade(player, kind)
	local profile = profiles[player]
	if not profile or not UPGRADES[kind] then return end
	local level = profile.upgrades[kind]
	if level >= MAX_UPGRADE_LEVEL then return end
	local cost = upgradeCost(kind, level)
	if profile.cash < cost then
		notify(player, "Not enough credits", "bad")
		return
	end
	profile.cash -= cost
	profile.upgrades[kind] = level + 1
	publish(player)
	remote:FireClient(player, "Upgraded", kind, level + 1)
end

remote.OnServerEvent:Connect(function(player, action, argument)
	local profile = profiles[player]
	if not profile or type(action) ~= "string" then return end
	if action == "Charge" then
		local now = os.clock()
		if profile.lastCharge and now - profile.lastCharge < CHARGE_COOLDOWN then return end
		profile.lastCharge = now
		local s = profile.stats
		profile.energy = math.min(s.capacity, profile.energy + s.charge)
		player:SetAttribute("Energy", profile.energy)
		if s.autoSell and profile.energy >= s.capacity then sell(player, true) end
	elseif action == "Sell" then
		sell(player, false)
	elseif action == "Upgrade" and type(argument) == "string" then
		upgrade(player, argument)
	elseif action == "Rebirth" then
		rebirth(player)
	elseif action == "Home" then
		teleportHome(player)
	end
end)

local function loadData(player)
	local data = nil
	local ok = true
	if store then
		ok = pcall(function()
			data = store:GetAsync("player_" .. player.UserId)
		end)
	end
	local profile = {
		cash = 0, energy = 0, rebirths = 0, items = {},
		upgrades = { Capacity = 0, Charge = 0, Sell = 0 },
		canSave = ok and store ~= nil,
	}
	if type(data) == "table" then
		profile.cash = tonumber(data.cash) or 0
		profile.rebirths = tonumber(data.rebirths) or 0
		if type(data.items) == "table" then
			for _, id in ipairs(data.items) do
				if ITEM_BY_ID[id] then profile.items[id] = true end
			end
		end
		if type(data.upgrades) == "table" then
			for kind in pairs(UPGRADES) do
				profile.upgrades[kind] = math.clamp(tonumber(data.upgrades[kind]) or 0, 0, MAX_UPGRADE_LEVEL)
			end
		end
	end
	return profile
end

local function saveData(player)
	local profile = profiles[player]
	if not profile or not profile.canSave then return end
	local items = {}
	for id in pairs(profile.items) do table.insert(items, id) end
	local data = {
		cash = math.floor(profile.cash),
		rebirths = profile.rebirths,
		items = items,
		upgrades = profile.upgrades,
	}
	pcall(function()
		store:SetAsync("player_" .. player.UserId, data)
	end)
end

local function onPlayerAdded(player)
	local profile = loadData(player)
	if not player.Parent then return end

	local plot = nil
	for _, candidate in ipairs(plots) do
		if not candidate.owner then
			plot = candidate
			break
		end
	end

	local leaderstats = Instance.new("Folder")
	leaderstats.Name = "leaderstats"
	local cashValue = Instance.new("NumberValue")
	cashValue.Name = "Credits"
	cashValue.Parent = leaderstats
	local rebirthValue = Instance.new("IntValue")
	rebirthValue.Name = "Rebirths"
	rebirthValue.Parent = leaderstats
	leaderstats.Parent = player
	profile.leaderCash = cashValue
	profile.leaderRebirths = rebirthValue

	profiles[player] = profile
	profile.stats = stats(profile)

	if not plot then
		notify(player, "Server is full of stations, try another server", "bad")
		publish(player)
		return
	end

	plot.owner = player
	profile.plot = plot
	plot.sign.Text = string.upper(player.DisplayName) .. "'S STATION"
	player:SetAttribute("PlotIndex", plot.index)
	for _, id in ipairs(ORDER) do
		if profile.items[id] then buildItem(plot, id, false) end
	end
	refreshButtons(plot)
	publish(player)

	player.CharacterAdded:Connect(function(character)
		character:WaitForChild("HumanoidRootPart", 10)
		task.wait(0.2)
		teleportHome(player)
	end)
	if player.Character then
		task.defer(teleportHome, player)
	end
end

Players.PlayerAdded:Connect(onPlayerAdded)
for _, player in ipairs(Players:GetPlayers()) do
	task.spawn(onPlayerAdded, player)
end

Players.PlayerRemoving:Connect(function(player)
	saveData(player)
	local profile = profiles[player]
	if profile and profile.plot then
		clearPlot(profile.plot)
		profile.plot.owner = nil
		profile.plot.sign.Text = "EMPTY DOCK"
	end
	profiles[player] = nil
end)

task.spawn(function()
	local blink = false
	while true do
		task.wait(1)
		blink = not blink
		for player, profile in pairs(profiles) do
			local s = profile.stats
			if s and s.rate > 0 then
				profile.energy = math.min(s.capacity, profile.energy + s.rate)
			end
			if s and s.autoSell and profile.energy >= s.capacity then
				sell(player, true)
			else
				publish(player)
			end
			local plot = profile.plot
			if plot then
				local ratio = s and s.capacity > 0 and math.clamp(profile.energy / s.capacity, 0, 1) or 0
				for _, fill in ipairs(plot.fills) do
					if fill.part.Parent then
						local height = math.max(0.1, fill.height * ratio)
						fill.part.Size = Vector3.new(4.2, height, 4.2)
						fill.part.CFrame = fill.bottom * CFrame.new(0, height / 2, 0)
						fill.part.Color = ratio >= 1 and RED or GREEN
					end
				end
				for _, light in ipairs(plot.blinkers) do
					if light.Parent then light.Transparency = blink and 0 or 0.8 end
				end
				for _, entry in ipairs(plot.buttonParts) do
					if entry.pad.Parent then
						local affordable = profile.cash >= entry.item.price
						entry.pad.Color = affordable and GREEN or RED
						entry.price.TextColor3 = affordable and GREEN or RED
					end
				end
			end
		end
	end
end)

		for _, s in ipairs(plot.spinners) do
			if s.part.Parent then
				s.part.CFrame = s.pivot * CFrame.Angles(s.speed.X * t, s.speed.Y * t, s.speed.Z * t) * s.offset
			end
		end
	end
end)
